# Read-only long-run health suite. Run in an entered campaign at a checkpoint.
# It deliberately does not require the ten countries to survive or NMG to
# remain a Mexican subject after the startup boundary.

last_date = 1900.1.1

tests = {
	ywc_longrun_no_negative_population = {
		acceptable_fail_rate = 0.0
		success = {
			NOT = {
				any_country = {
					total_population < 0
				}
			}
		}
		fail = { game_date > 1900.1.1 }
	}

	ywc_longrun_no_orphaned_capital = {
		acceptable_fail_rate = 0.0
		success = {
			NOT = {
				any_country = {
					is_country_alive = yes
					NOT = {
						capital = { exists = yes }
					}
				}
			}
		}
		fail = { game_date > 1900.1.1 }
	}

	ywc_longrun_shared_variables_bounded = {
		acceptable_fail_rate = 0.0
		success = {
			c:SHU ?= {
				var:ywc_heritage_legitimacy >= 0
				var:ywc_heritage_legitimacy <= 100
				var:ywc_maritime_network_level >= 0
				var:ywc_maritime_network_level <= 100
				var:ywc_autonomy_pressure >= 0
				var:ywc_autonomy_pressure <= 100
			}
		}
		fail = { game_date > 1900.1.1 }
	}

	ywc_longrun_content_health = {
		acceptable_fail_rate = 0.0
		success = {
			c:SHU ?= {
				has_journal_entry = ywc_je_dlc_compatibility
			}
		}
		fail = { game_date > 1900.1.1 }
	}
}
