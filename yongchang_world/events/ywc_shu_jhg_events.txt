    option = {
        name = ywc_shu.4.a
        default_option = yes
        ai_chance = {
            base = 50

            modifier = { trigger = { var:ywc_autonomy_pressure >= 70 } add = -18 }

        trigger = {
            NOT = { has_variable = ywc_route_shu_bureaucratic_integration_active }

        if = {
            limit = { has_variable = ywc_route_shu_bureaucratic_integration_abandon }
            remove_variable = ywc_route_shu_bureaucratic_integration_abandon
        }
        set_variable = { name = ywc_route_shu_bureaucratic_integration_active value = 1 }
        if = {
            limit = { var:ywc_flavor_shu_court_balance >= 70 }
            add_modifier = { name = ywc_flavor_shu_high months = 12 }
        }
        if = {
            limit = { var:ywc_flavor_shu_court_balance < 30 }
            add_modifier = { name = ywc_flavor_shu_low months = 12 }
        }
        set_variable = { name = ywc_route_shu_bureaucratic_integration_progress value = 0 }
        change_variable = { name = ywc_route_shu_bureaucratic_integration_progress add = 20 }
        clamp_variable = { name = ywc_route_shu_bureaucratic_integration_progress min = 0 max = 100 }
        ywc_add_heritage_legitimacy = yes
        add_modifier = { name = ywc_shu_bureaucratic_commitment months = 12 }
    }
    option = {
        name = ywc_shu.4.b
        ai_chance = {
            base = 80
            modifier = { trigger = { var:ywc_route_shu_bureaucratic_integration_progress >= 80 } add = 15 }
            modifier = { trigger = { var:ywc_heritage_legitimacy >= 60 } add = 12 }
        }
        trigger = {
            has_variable = ywc_route_shu_bureaucratic_integration_active
            has_variable = ywc_route_shu_bureaucratic_integration_progress
        }
        if = {
            limit = { var:ywc_route_shu_bureaucratic_integration_progress >= 100 }
            set_variable = { name = ywc_route_shu_bureaucratic_integration_success value = 1 }
            if = {
                limit = { var:ywc_flavor_shu_court_balance >= 70 }
                add_modifier = { name = ywc_flavor_shu_high months = 24 }
            }
            remove_variable = ywc_route_shu_bureaucratic_integration_active
            ywc_add_heritage_legitimacy = yes
            add_modifier = { name = ywc_shu_bureaucratic_commitment months = 120 }
        }
        else = {
            change_variable = { name = ywc_route_shu_bureaucratic_integration_progress add = 20 }
            clamp_variable = { name = ywc_route_shu_bureaucratic_integration_progress min = 0 max = 100 }
            ywc_add_heritage_legitimacy = yes
        }
    }
    option = {
        name = ywc_shu.4.c
        ai_chance = {
            base = 4
            modifier = { trigger = { var:ywc_route_shu_bureaucratic_integration_progress >= 60 } add = 38 }
            modifier = { trigger = { var:ywc_heritage_legitimacy < 30 } add = 15 }
