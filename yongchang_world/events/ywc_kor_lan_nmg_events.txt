ywc_kor.5 = {
    type = country_event
    title = ywc_kor.5.t
    desc = ywc_kor.5.d
    duration = 1
    option = {
        name = ywc_kor.5.a
        default_option = yes
        trigger = {
            NOT = { has_variable = ywc_route_kor_small_china_active }
            NOT = { has_variable = ywc_route_kor_national_foundation_active }
            NOT = { has_variable = ywc_route_kor_small_china_success }
            NOT = { has_variable = ywc_route_kor_small_china_failure }
            NOT = { has_variable = ywc_route_kor_small_china_abandon }
        }
        set_variable = { name = ywc_route_kor_small_china_active value = 1 }
        set_variable = { name = ywc_route_kor_small_china_progress value = 0 }
        change_variable = { name = ywc_route_kor_small_china_progress add = 20 }
        ywc_add_heritage_legitimacy = yes
        add_modifier = { name = ywc_kor_small_china months = 12 }
    }
    option = {
        name = ywc_kor.5.b
        trigger = { has_variable = ywc_route_kor_small_china_active }
        if = {
            limit = { var:ywc_route_kor_small_china_progress >= 100 }
            set_variable = { name = ywc_route_kor_small_china_success value = 1 }
            remove_variable = ywc_route_kor_small_china_active
            add_modifier = { name = ywc_kor_small_china months = 120 }
        }
        else = {
            change_variable = { name = ywc_route_kor_small_china_progress add = 20 }
            clamp_variable = { name = ywc_route_kor_small_china_progress min = 0 max = 100 }
            ywc_add_heritage_legitimacy = yes
        }
    }
    option = {
        name = ywc_kor.5.c
        trigger = { has_variable = ywc_route_kor_small_china_active }
        if = {
            limit = { var:ywc_route_kor_small_china_progress >= 60 }
            set_variable = { name = ywc_route_kor_small_china_failure value = 1 }
            add_modifier = { name = ywc_route_failure_backlash months = 24 }
        }
        else = {
            set_variable = { name = ywc_route_kor_small_china_abandon value = 1 }
            change_variable = { name = ywc_heritage_legitimacy add = -10 }
            clamp_variable = { name = ywc_heritage_legitimacy min = 0 max = 100 }
        }
        remove_variable = ywc_route_kor_small_china_active
    }
}
