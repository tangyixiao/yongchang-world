    }
    option = {
        name = ywc_oir.5.b
        trigger = { has_variable = ywc_route_oir_bureaucratic_khanate_active }

            limit = { var:ywc_route_oir_bureaucratic_khanate_progress >= 100 }
            set_variable = { name = ywc_route_oir_bureaucratic_khanate_success value = 1 }
            remove_variable = ywc_route_oir_bureaucratic_khanate_active
            add_modifier = { name = ywc_oir_bureaucratic_khanate months = 120 }
        }
        else = {
            change_variable = { name = ywc_route_oir_bureaucratic_khanate_progress add = 20 }
            clamp_variable = { name = ywc_route_oir_bureaucratic_khanate_progress min = 0 max = 100 }
            ywc_add_heritage_legitimacy = yes
        }
    }
    option = {
        name = ywc_oir.5.c
        trigger = { has_variable = ywc_route_oir_bureaucratic_khanate_active }
        if = {
            limit = { var:ywc_route_oir_bureaucratic_khanate_progress >= 60 }
            set_variable = { name = ywc_route_oir_bureaucratic_khanate_failure value = 1 }
            add_modifier = { name = ywc_route_failure_backlash months = 24 }

        else = {
            set_variable = { name = ywc_route_oir_bureaucratic_khanate_abandon value = 1 }
            change_variable = { name = ywc_heritage_legitimacy add = -10 }
            clamp_variable = { name = ywc_heritage_legitimacy min = 0 max = 100 }
