# Crisis C07 债务、矿权与公法: scripted triggers. All state variables are
# per-country with the ywc_c07_ prefix; every cross-country read stays
# inside an explicit c:XXX ?= { } block so no variable is read outside its
# writer's scope. Roles: 1 = host (asset/land location), 2 = funder (real
# money side). Stage values: 1600/1601 host investigation, 1602 funder
# guarantee answer, 1603 host labor terms, 1604 host road order, 1605
# funder reschedule, 1606 host mine liability (only with real mining-state
# labor risk), 1607 tariff answers (host and funder sides of one event),
# 1608 host two-phase proposal, 1609 exit ledger (both sides), 1610
# settlement (both sides), 1611 aftermath (both sides); internal waiting
# stages: 1612 (funder awaits the guarantee round), 1613 (host awaits the
# funder's guarantee answer), 1614 (host awaits the reschedule answer),
# 1615 (funder awaits the road/liability phase), 1616 (host awaits the
# funder's tariff answer), 1617 (funder awaits the host's tariff routing),
# 1618 (funder awaits the exit-ledger round), 1619 (host awaits the funder's
# exit-ledger answer), 1621 (funder awaits settlement routing), 1620
# (terminal aftermath marker).
# Producer facts read here (each inside the holder's own scope):
# - registered mine buildings (building_iron_mine/coal/lead/gold/sulfur)
# - in_default / declared_bankruptcy / gold_reserves (fiscal distress)
# - turmoil and radical laborers in a mining state (labor/land conflict)
# - financial districts or trade centers and gold_reserves (funder money)

# --- Holder tag group -------------------------------------------------------

ywc_c07_is_pool_tag = { OR = { c:LAN ?= this c:WBK ?= this c:SHU ?= this c:KOR ?= this c:DMG ?= this c:NMG ?= this } }
ywc_c07_is_host = { var:ywc_c07_role ?= 1 }
ywc_c07_is_funder = { var:ywc_c07_role ?= 2 }

# --- Real local facts (evaluated at the holder's own scope) -----------------

ywc_c07_mining_development = { any_scope_state = { any_scope_building = { OR = { is_building_type = building_iron_mine is_building_type = building_coal_mine is_building_type = building_lead_mine is_building_type = building_gold_mine is_building_type = building_sulfur_mine } } } }

ywc_c07_funder_money = { OR = { gold_reserves >= 1000 any_scope_state = { any_scope_building = { OR = { is_building_type = building_financial_district is_building_type = building_trade_center } } } } }
# The case opens only when debt/fiscal distress co-occurs with a real
# development-labour/land conflict on registered mining ground.
ywc_c07_host_conditions = { ywc_c07_mining_development = yes ywc_c07_fiscal_distress = yes ywc_c07_labor_risk = yes }
