# Crisis C05 太平洋自治争议: scripted triggers. All state variables are
# per-country with the ywc_c05_ prefix; every cross-country read stays inside
# an explicit c:XXX ?= { } block so no variable is read outside its writer's
# scope. Stage values: 1400/1401 NMG investigation, 1402 MEX answer,
# 1403 third-country answer, 1404 NMG payroll, 1405 MEX appointment,
# 1406 NMG wharf deeds, 1407 NMG remittance explanation, 1408 NMG draft,
# 1409 MEX veto, 1412 NMG aftermath (event ywc_crisis.1411), 1414 MEX
# remittance verdict (event ywc_crisis.1407), 1415 NMG confirm and 1416 MEX
# confirm (event ywc_crisis.1410); internal waiting stages: 1413 (NMG awaits
# answers), 1417 (NMG awaits the remittance verdict), 1418 (MEX awaits the
# renegotiation round), 1419 (terminal aftermath marker), 1420 (NMG awaits the
# federal veto).

# --- Entry readiness ---------------------------------------------------------

# A real dispute exists only when a settled autonomy-revenue result (the local
# remittance quota was actually negotiated) or the old campaign crisis left a
# recorded outcome. An independent NMG with neither marker gets ordinary
# country content instead of a fabricated return to the federal bargain.
ywc_c05_fiscal_dispute = { OR = { AND = { var:ywc_v03_nmg_v03_nmg_autonomy_revenue_settled ?= 1 OR = { var:ywc_v03_nmg_v03_nmg_autonomy_revenue_outcome ?= 1 var:ywc_v03_nmg_v03_nmg_autonomy_revenue_outcome ?= 2 var:ywc_v03_nmg_v03_nmg_autonomy_revenue_outcome ?= 3 } } has_variable = ywc_campaign_crisis5_outcome } }

ywc_c05_case_ready_nmg = { c:NMG ?= this is_country_alive = yes has_variable = ywc_v03_initialized ywc_v03_has_capacity = yes ywc_c05_fiscal_dispute = yes c:MEX ?= { is_country_alive = yes } NOT = { has_variable = ywc_c05_started } NOT = { has_variable = ywc_c05_outcome } }

# --- Actual contact links (adjacency, diplomatic contact, market, port) ------
# Evaluated at the third country's own scope, so they enter from any reader
# through c:XXX ?= { } instead of comparing tags against this.

ywc_c05_market_nmg = { c:NMG ?= { is_country_alive = yes } is_in_same_market_area = c:NMG.capital }
ywc_c05_contact_nmg = { OR = { any_scope_state = { any_neighbouring_state = { c:NMG ?= owner } } has_diplomatic_relevance = c:NMG ywc_c05_market_nmg = yes AND = { has_port_country = yes ywc_c05_market_nmg = yes } } }

ywc_c05_usa_available_nmg = { c:USA ?= { is_country_alive = yes ywc_c05_contact_nmg = yes } }
ywc_c05_shu_available_nmg = { c:SHU ?= { is_country_alive = yes ywc_c05_contact_nmg = yes } }
ywc_c05_jhg_available_nmg = { c:JHG ?= { is_country_alive = yes ywc_c05_contact_nmg = yes } }
ywc_c05_nqg_available_nmg = { c:NQG ?= { is_country_alive = yes ywc_c05_contact_nmg = yes } }
