# Crisis C08 侨民保护与沿岸裁判: scripted triggers. All state variables are
# per-country with the ywc_c08_ prefix; every cross-country read stays inside
# an explicit c:XXX ?= { } block so no variable is read outside its writer's
# scope. Stage values: 1700/1701 DMG investigation, 1702 NMG note, 1703 DMG
# court answer, 1704 shared funding, 1705 DMG residence, 1706 DMG protest,
# 1707 NMG conditions, 1716 DMG conditions reply (same event ywc_crisis.1707),
# 1708 DMG forum proposal, 1709 shared resident choice, 1710 shared settlement
# confirm, 1711 shared aftermath (DMG via journal 3, partners via journal 1);
# internal waiting stages: 1712 (DMG awaits the external note), 1715 (DMG
# awaits the conditions answer), 1718 (NMG awaits the forum proposal),
# 1721 (NMG awaits the DMG settlement), 1719 (terminal aftermath marker).

# --- Entry readiness ---------------------------------------------------------
# The case opens only on real links between the port jurisdiction (DMG) and a
# living external claimant (NMG): market, adjacency, diplomatic relevance or a
# port-to-port trade lane. Cultural or religious similarity alone never
# qualifies, and no third country joins without its own real link.

ywc_c08_market_dmg = { c:DMG ?= { is_country_alive = yes } is_in_same_market_area = c:DMG.capital }

ywc_c08_contact_dmg = { OR = { any_scope_state = { any_neighbouring_state = { c:DMG ?= owner } } has_diplomatic_relevance = c:DMG ywc_c08_market_dmg = yes AND = { has_port_country = yes ywc_c08_market_dmg = yes } } }

ywc_c08_case_ready_dmg = { c:DMG ?= this is_country_alive = yes has_variable = ywc_v03_initialized ywc_v03_has_capacity = yes c:NMG ?= { is_country_alive = yes ywc_c08_contact_dmg = yes } NOT = { has_variable = ywc_c08_started } NOT = { has_variable = ywc_c08_outcome } }
