# Local v0.3 SHU trigger vocabulary, limited to vanilla-supported country/state scopes.
ywc_shu_v03_has_pilot_state = {
	OR = {
		AND = { var:ywc_shu_v03_pilot_state = 1 any_scope_state = { owner = ROOT state_region = s:STATE_SICHUAN } }
		AND = { var:ywc_shu_v03_pilot_state = 2 any_scope_state = { owner = ROOT state_region = s:STATE_CHONGQING } }
		AND = { var:ywc_shu_v03_pilot_state = 3 any_scope_state = { owner = ROOT state_region = s:STATE_YUNNAN } }
		AND = { var:ywc_shu_v03_pilot_state = 4 any_scope_state = { owner = ROOT state_region = s:STATE_GUIZHOU } }
	}
}

		has_law_or_variant = law_type:law_land_based_taxation

		has_law_or_variant = law_type:law_proportional_taxation
