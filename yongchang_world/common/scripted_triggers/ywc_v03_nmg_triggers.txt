# Fixed remittances bind the actual MEX overlord and the exact accepted terms.
ywc_nmg_mexshare_slot_available = {
	NOT = { var:ywc_nmg_mexshare_state ?= 1 }
	NOT = { var:ywc_nmg_mexshare_state ?= 2 }
	OR = {
		NOT = { has_variable = ywc_nmg_mexshare_contract_generation }
		AND = { var:ywc_nmg_mexshare_ended_generation ?= var:ywc_nmg_mexshare_contract_generation has_variable = ywc_nmg_mexshare_paid_total var:ywc_nmg_mexshare_earned_total ?= var:ywc_nmg_mexshare_paid_total }
	}
}

ywc_nmg_mexshare_link_intact = {
	c:NMG ?= this
	is_country_alive = yes
	is_subject = yes
	exists = overlord
	exists = c:MEX
	is_subject_of = c:MEX
	overlord ?= c:MEX
	c:MEX = { is_country_alive = yes relations:c:NMG >= 0 }
}

ywc_nmg_mexshare_can_offer = {
	ywc_nmg_mexshare_link_intact = yes
	ywc_nmg_revenue_fixed_ready = yes
	has_variable = ywc_nmg_revenue_quota
	ywc_nmg_mexshare_slot_available = yes
}

ywc_nmg_mexshare_offer_matches = {
	exists = scope:ywc_nmg_mexshare_requester
	exists = scope:ywc_nmg_mexshare_request_partner
	scope:ywc_nmg_mexshare_request_partner ?= this
	scope:ywc_nmg_mexshare_requester ?= c:NMG
	scope:ywc_nmg_mexshare_requester = {
		ywc_nmg_mexshare_link_intact = yes
		var:ywc_nmg_mexshare_partner ?= root
		var:ywc_nmg_mexshare_state ?= 1
		has_variable = ywc_nmg_mexshare_request_timer
		var:ywc_nmg_mexshare_generation ?= scope:ywc_nmg_mexshare_request_generation
		var:ywc_nmg_mexshare_serial ?= scope:ywc_nmg_mexshare_request_serial
		var:ywc_nmg_mexshare_quote_serial ?= scope:ywc_nmg_mexshare_request_serial
		var:ywc_nmg_mexshare_quote_monthly ?= scope:ywc_nmg_mexshare_request_monthly
		var:ywc_nmg_mexshare_quote_term ?= scope:ywc_nmg_mexshare_request_term
		var:ywc_nmg_mexshare_quote_right ?= scope:ywc_nmg_mexshare_request_right
	}
}

ywc_nmg_mexshare_authorizable = {
	ywc_nmg_mexshare_offer_matches = yes
	c:MEX ?= this

	relations:scope:ywc_nmg_mexshare_requester >= 0

	has_variable = ywc_nmg_mexshare_contract_generation
