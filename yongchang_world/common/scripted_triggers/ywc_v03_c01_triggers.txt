# Crisis C01 天下名分与宗藩再议: scripted triggers. All state variables are
# per-country with the ywc_c01_ prefix; every cross-country read stays inside
# an explicit c:XXX ?= { } block so no variable is read outside its writer's
# scope. Stage values: 1000/1001 initiator investigation, 1002 JHG response,
# 1003 KOR/DMG response, 1004 funding, 1005 authorization, 1006 port tariff,
# 1007 SHU levy proposal, 1012 demanded-party levy answer (event ywc_crisis.1007),
# 1008 renegotiation, 1009 non-signatory choice, 1010 settlement, 1011 review;
# internal waiting stages: 1013 (initiator awaits responses), 1014 (non-signatory
# awaits renegotiation), 1015 (signatory awaits settlement), 1017 (initiator
# awaits levy answer), 1018 (responder awaits the opening of responses).

# --- Entry readiness ---------------------------------------------------------

# SHU carries a settled fiscal or customs reform that created a real claim
# (central audit / provincial share, or unified customs / negotiated share).
ywc_c01_shu_reform_claim = { OR = { AND = { var:ywc_v03_shu_v03_shu_fiscal_register_settled ?= 1 OR = { var:ywc_v03_shu_v03_shu_fiscal_register_outcome ?= 1 var:ywc_v03_shu_v03_shu_fiscal_register_outcome ?= 2 } } AND = { var:ywc_v03_shu_v03_shu_port_treasury_settled ?= 1 OR = { var:ywc_v03_shu_v03_shu_port_treasury_outcome ?= 1 var:ywc_v03_shu_v03_shu_port_treasury_outcome ?= 2 } } } }

# JHG carries a settled tribute or court-assembly result, so a promised
# contribution exists to clash with the newly claimed rights.
ywc_c01_jhg_prior_settlement = { OR = { var:ywc_jhg_v03_jhg_tribute_autonomy_settled ?= 1 var:ywc_jhg_v03_jhg_court_assembly_settled ?= 1 } }

ywc_c01_case_ready_shu = { c:SHU ?= this is_country_alive = yes has_variable = ywc_v03_initialized ywc_v03_has_capacity = yes ywc_c01_shu_reform_claim = yes c:JHG ?= { is_country_alive = yes ywc_c01_jhg_prior_settlement = yes } NOT = { has_variable = ywc_c01_started } NOT = { has_variable = ywc_c01_outcome } }

ywc_c01_case_ready_jhg = { c:JHG ?= this is_country_alive = yes has_variable = ywc_v03_initialized ywc_v03_has_capacity = yes ywc_c01_jhg_prior_settlement = yes NOT = { c:SHU ?= { is_country_alive = yes } } OR = { ywc_c01_kor_available_jhg = yes ywc_c01_dmg_available_jhg = yes } NOT = { has_variable = ywc_c01_started } NOT = { has_variable = ywc_c01_outcome } }

# --- Actual contact links (adjacency, diplomatic contact, market, port) ------

ywc_c01_adjacent_shu = { any_scope_state = { any_neighbouring_state = { c:SHU ?= owner } } }
ywc_c01_market_shu = { c:SHU ?= { is_country_alive = yes } is_in_same_market_area = c:SHU.capital }
ywc_c01_contact_shu = { OR = { ywc_c01_adjacent_shu = yes has_diplomatic_relevance = c:SHU ywc_c01_market_shu = yes AND = { has_port_country = yes ywc_c01_market_shu = yes } } }
ywc_c01_adjacent_jhg = { any_scope_state = { any_neighbouring_state = { c:JHG ?= owner } } }
ywc_c01_market_jhg = { c:JHG ?= { is_country_alive = yes } is_in_same_market_area = c:JHG.capital }
ywc_c01_contact_jhg = { OR = { ywc_c01_adjacent_jhg = yes has_diplomatic_relevance = c:JHG ywc_c01_market_jhg = yes AND = { has_port_country = yes ywc_c01_market_jhg = yes } } }

# Initiator-side availability: does an eligible responder exist at all? These
# are evaluated at the initiator's scope, so they enter the responder scope
# with c:XXX ?= { } instead of comparing tags against this.
