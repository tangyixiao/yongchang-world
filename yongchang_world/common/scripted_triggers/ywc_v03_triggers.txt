# Shared gate for a newly auto-activated v0.3 arc. The JE must also declare
# its current country tag and arc-specific started/settled checks.
ywc_v03_has_capacity = {
	has_variable = ywc_v03_initialized
	var:ywc_v03_release_version >= 3
	var:ywc_v03_active_arc_count < 3
	var:ywc_v03_visible_task_count < 6
}

# Shared cooldown gate for deliberate country-arc event delivery.
ywc_v03_can_deliver_major = {
	NOT = { has_variable = ywc_v03_major_event_cooldown }
}
