# Yongchang World - Han-royal appearance gate
# Used by gfx/portraits/portrait_modifiers/ywc_han_royal_appearance.txt.
# Context: game/common/scripted_triggers/00_clothes_triggers.txt:1121
# (chinese_imperial_clothes_trigger) gives every Han-heritage monarch of a
# monarchy the Qing-imperial outfit and cap.  This trigger narrows that override
# to the three Han royal houses only, so vanilla CHI and every other Han
# monarchy keep the vanilla look.

ywc_han_royal_appearance_trigger = {
	exists = scope:character
	scope:character = {
		is_ruler_of_own_country = yes
		exists = home_country
		home_country = {
			OR = {
				c:SHU ?= this
				c:DMG ?= this
				c:NMG ?= this
			}
		}
	}
}
