# Royal identity is independent of whether the local fallback is enabled.
ywc_han_royal_appearance_trigger = {
	exists = scope:character
	scope:character = {
		OR = {
			is_ruler_of_own_country = yes
			is_heir_of_own_country = yes
		}
		culture = {
			has_discrimination_trait = heritage_han
		}
		exists = home_country
		home_country = {
			OR = {
				c:SHU ?= this
				c:DMG ?= this
				c:NMG ?= this
				c:MHG ?= this
			}
			OR = {
				has_law = law_type:law_monarchy
				has_law = law_type:law_social_monarchy
			}
		}
	}
}

# Use the vanilla court fallback only when the optional EACR companion is absent.
ywc_han_royal_fallback_trigger = {
	EACR_is_active_trigger = no
	ywc_han_royal_appearance_trigger = yes
}

# Explicit replacement is required because this folder contains scripted
# trigger REPLACE directives. Keep the vanilla scope and rule intact, excluding
# only royals covered by the identity gate above.
REPLACE:chinese_imperial_clothes_trigger = {
	exists = scope:character
	scope:character = {
		is_ruler_of_own_country = yes
		exists = home_country
		home_country = {
			OR = {
				has_law = law_type:law_monarchy
				has_law = law_type:law_social_monarchy
			}
		}
		OR = {
			culture = {
				has_discrimination_trait = heritage_han
			}
			AND = {
				exists = c:CHI
				home_country = c:CHI
			}
		}
		NOT = { ywc_han_royal_appearance_trigger = yes }
	}
}

# Keep all vanilla court-clothes exclusions and culture checks, adding only the
# same narrow exclusion so royal fallback candidates cannot select court caps.
REPLACE:chinese_court_clothes_trigger = {
	exists = scope:character
	scope:character = {
		NOT = { ywc_han_royal_appearance_trigger = yes }
		NOT = {
			AND = {
				exists = home_country
				is_ruler_of_own_country = yes
				culture = {
					has_discrimination_trait = heritage_tai
					has_discrimination_trait_group = heritage_group_southeast_asian
				}
				home_country = {
					OR = {
						has_law = law_type:law_monarchy
						has_law = law_type:law_social_monarchy
						has_law_or_variant = law_type:law_theocracy
					}
				}
			}
		}
		culture = {
			NOT = { this = cu:kalmyk }
			NOT = { this = cu:korean }
			OR = {
				has_discrimination_trait = tradition_sinosphere
				has_discrimination_trait = heritage_han
				has_discrimination_trait = language_tibetan
				has_discrimination_trait_group = language_group_mongolic
				has_discrimination_trait = heritage_tai
				has_discrimination_trait = language_hmong
			}
		}
	}
}
