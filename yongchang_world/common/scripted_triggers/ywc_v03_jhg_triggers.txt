# JHG foundation scripted triggers. Owned by country_jhg.

ywc_jhg_finance_safe = {
	bureaucracy >= 0
	net_fixed_income >= 0
	NOT = { in_default = yes }
	NOT = { has_modifier = declared_bankruptcy }
	NOT = { is_at_war = yes }
}


ywc_jhg_court_limited_route_execution_ready = {
	OR = {
		AND = { var:ywc_jhg_court_route = 1 bureaucracy >= 40 government_legitimacy >= 20 any_scope_state = { owner = ROOT any_scope_building = { is_building_type = building_government_administration occupancy >= 0.6 building_has_goods_shortage = no } } }
		AND = { var:ywc_jhg_court_route = 2 ywc_jhg_court_property_rep = yes government_legitimacy >= 20 }
		AND = { var:ywc_jhg_court_route = 3 ig:ig_industrialists ?= { is_powerful = yes } }
	}
}
ywc_jhg_convoy_limited_route_execution_ready = {
	OR = {
		AND = { var:ywc_jhg_convoy_route = 1 navy_size >= 3 any_scope_state = { owner = ROOT any_scope_building = { is_building_type = building_port occupancy >= 0.6 building_has_goods_shortage = no } } }
		AND = { var:ywc_jhg_convoy_route = 2 navy_size >= 3 ywc_jhg_escort_watch_ready = yes }
		AND = { var:ywc_jhg_convoy_route = 3 bureaucracy >= 10 }
	}
}
ywc_jhg_tribute_limited_route_execution_ready = {
	OR = {
		AND = { var:ywc_jhg_tribute_route = 1 ywc_jhg_tribute_relationship_open = yes bureaucracy >= 10 }
		AND = { var:ywc_jhg_tribute_route = 2 ywc_jhg_share_contract_active = yes }
		AND = { var:ywc_jhg_tribute_route = 3 OR = { is_subject = no ywc_jhg_tribute_relationship_open = yes } government_legitimacy >= 20 }
	}
}
# --- v03_jhg_court_assembly ------------------------------------------------

ywc_jhg_court_bureaucracy_ready = {
	bureaucracy >= 40
}

ywc_jhg_court_administration_staffed = {
	any_scope_state = {
		owner = ROOT
		any_scope_building = {
			is_building_type = building_government_administration
			occupancy >= 0.8
			building_has_goods_shortage = no
		}
	}
}

ywc_jhg_court_property_rep = {
	OR = {
		has_law_or_variant = law_type:law_landed_voting
		has_law_or_variant = law_type:law_wealth_voting
		has_law_or_variant = law_type:law_census_voting
		has_law_or_variant = law_type:law_oligarchy
		ig:ig_landowners ?= { is_powerful = yes }
	}
}

ywc_jhg_court_route_one_ready = {
	AND = {
		var:ywc_jhg_court_route = 1
		ywc_jhg_court_bureaucracy_ready = yes
		ywc_jhg_court_administration_staffed = yes
		government_legitimacy >= 40
	}
}

ywc_jhg_court_route_two_ready = {
	AND = {
		var:ywc_jhg_court_route = 2
		ywc_jhg_court_property_rep = yes
		government_legitimacy >= 30
	}
}

ywc_jhg_court_route_three_ready = {
	AND = {
		var:ywc_jhg_court_route = 3
		ig:ig_industrialists ?= { is_powerful = yes }
		NOT = { in_default = yes }
	}
}

ywc_jhg_court_route_execution_ready = {
	AND = {
		ywc_jhg_finance_safe = yes
		OR = {
		ywc_jhg_court_route_one_ready = yes
		ywc_jhg_court_route_two_ready = yes
		ywc_jhg_court_route_three_ready = yes
	}	}
}


ywc_jhg_court_can_pay = {

}
