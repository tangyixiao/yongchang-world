# Crisis C06 滇缅山口与双重贡赋: scripted triggers. All state variables are
# per-country with the ywc_c06_ prefix; every cross-country read stays inside
# an explicit c:XXX ?= { } block so no variable is read outside its writer's
# scope. Stage values: 1500/1501 proposer investigation, 1502 claimant-A reply
# (SHU, or the actual demanding party when SHU is gone), 1503 claimant-B reply
# (Burma, or another real neighbouring hill country), 1504 shares, 1505 council
# mandate, 1506 transit demand and 1512 the local answer side of the same event
# (ywc_crisis.1506), 1507 seal conflict, 1508 renegotiation, 1509 hill
# reservations, 1510 settlement, 1511 aftermath review; internal waiting stages:
# 1513 (proposer awaits replies), 1514 (non-signatory awaits renegotiation),
# 1515 (signatory awaits settlement), 1516 (aftermath terminal), 1518
# (respondent awaits the opening of replies).
# Producer variables read here (each inside its own country's block):
# - ywc_frontier_ktg/waa/kch/dli/ahm_resolved (settled frontier-start lines)
# - ywc_sw_shd/ljg_resolved   (settled southwest six-country line)
# - ywc_v03_shu_v03_shu_fiscal_register_* / ywc_v03_shu_v03_shu_provincial_*
#   and ywc_shu_prov_outcome (SHU settled border claims, read inside c:SHU)

# --- Holder tag groups -------------------------------------------------------

ywc_c06_is_pool_tag = { OR = { c:KTG ?= this c:WAA ?= this c:KCH ?= this c:SHD ?= this c:LJG ?= this c:DLI ?= this c:AHM ?= this } }
ywc_c06_is_participant = { OR = { ywc_c06_is_pool_tag = yes c:SHU ?= this c:BUR ?= this } }

# --- The two orders (evaluated at the local holder's own scope) --------------
# SHU-side order: a settled SHU border claim (fiscal register or provincial
# contract) plus a real contact link. Burma-side order: Burma alive with a real
# contact link over the passes.

ywc_c06_shu_border_claim = { c:SHU ?= { is_country_alive = yes OR = { AND = { var:ywc_v03_shu_v03_shu_fiscal_register_settled ?= 1 OR = { var:ywc_v03_shu_v03_shu_fiscal_register_outcome ?= 1 var:ywc_v03_shu_v03_shu_fiscal_register_outcome ?= 2 } } AND = { var:ywc_v03_shu_v03_shu_provincial_contract_settled ?= 1 var:ywc_shu_prov_outcome > 0 } } } }
ywc_c06_shu_order_ready = { ywc_c06_shu_border_claim = yes ywc_c06_contact_shu = yes }
ywc_c06_bur_order_ready = { c:BUR ?= { is_country_alive = yes } ywc_c06_contact_bur = yes }

# --- Priority gating for the single local proposer ---------------------------
# A pool country may open the case only when no higher-priority pool country is
# simultaneously able to open it (alive, initialized, with capacity and its own
# settled prior line), so exactly one proposer exists in any given month.

ywc_c06_ktg_able = { c:KTG ?= { is_country_alive = yes has_variable = ywc_v03_initialized ywc_v03_has_capacity = yes var:ywc_frontier_ktg_resolved ?= 1 } }
ywc_c06_waa_able = { c:WAA ?= { is_country_alive = yes has_variable = ywc_v03_initialized ywc_v03_has_capacity = yes var:ywc_frontier_waa_resolved ?= 1 } }
ywc_c06_kch_able = { c:KCH ?= { is_country_alive = yes has_variable = ywc_v03_initialized ywc_v03_has_capacity = yes var:ywc_frontier_kch_resolved ?= 1 } }
ywc_c06_shd_able = { c:SHD ?= { is_country_alive = yes has_variable = ywc_v03_initialized ywc_v03_has_capacity = yes var:ywc_sw_shd_resolved ?= 1 } }
ywc_c06_ljg_able = { c:LJG ?= { is_country_alive = yes has_variable = ywc_v03_initialized ywc_v03_has_capacity = yes var:ywc_sw_ljg_resolved ?= 1 } }

# --- Cross-country state helpers ---------------------------------------------
# Claimant markers on the proposer: ywc_c06_claimant_a 1 = SHU, 2 = BUR;
# ywc_c06_claimant_b 2 = BUR, 3 = KTG, 4 = WAA, 5 = KCH, 6 = SHD, 7 = LJG,
# 8 = DLI, 9 = AHM.

ywc_c06_claimant_a_alive = { OR = { AND = { var:ywc_c06_claimant_a ?= 1 c:SHU ?= { is_country_alive = yes } } AND = { var:ywc_c06_claimant_a ?= 2 c:BUR ?= { is_country_alive = yes } } } }
ywc_c06_claimant_a_gone = { NOT = { ywc_c06_claimant_a_alive = yes } }
# Claimant-B liveness is read inside ywc_c06_replies_resolved (dead or absent
# claimant B counts as resolved); no separate gone-trigger is needed.
