# Theme T12 "Shared Burden Demands Shared Voice": shared partner and phase guards.
# SHU is the compact court (JE owner); DMG is the sea-lane ally (trade party).

# Partner reachability: DMG must be alive, still diplomatically relevant to
# SHU, and hold a real port or trade-center stake in the lane it defends.
# Usable from SHU scope and from inside c:SHU blocks.
ywc_t12_partner_current = {
 c:DMG ?= {
  is_country_alive = yes
  has_diplomatic_relevance = c:SHU
  OR = {
   any_scope_state = { owner = c:DMG any_scope_building = { is_building_type = building_port occupancy >= 0.3 } }
   any_scope_state = { owner = c:DMG any_scope_building = { is_building_type = building_trade_center occupancy >= 0.3 } }
  }
 }
}

# Entry: political participation formed (a voting or assembly law) with a
# working legitimacy floor, public finance formed (staffed administration and
# a non-deficit budget), and the real cooperative relationship above.
ywc_t12_local_participation = {
 OR = {
  has_law_or_variant = law_type:law_landed_voting
  has_law_or_variant = law_type:law_wealth_voting
  has_law_or_variant = law_type:law_census_voting
  has_law_or_variant = law_type:law_universal_suffrage
  has_law_or_variant = law_type:law_right_of_assembly
  has_law_or_variant = law_type:law_protected_speech
 }
 legitimacy >= 25
}
ywc_t12_public_finance = {
 any_scope_state = { owner = ROOT any_scope_building = { is_building_type = building_government_administration occupancy >= 0.5 } }
 net_fixed_income >= 0
}
ywc_t12_entry_ready = {
 c:SHU ?= this
 ywc_t12_local_participation = yes
 ywc_t12_public_finance = yes
 ywc_t12_partner_current = yes
}

# Real war or mobilization: current war, mass conscription or militia law, or
# heavily staffed barracks. War is never an entry requirement; it only opens
# the returning-soldiers stage.
ywc_t12_has_mobilization = {
 OR = {
  is_at_war = yes
  has_law_or_variant = law_type:law_mass_conscription
  has_law_or_variant = law_type:law_national_militia
  any_scope_state = { owner = ROOT any_scope_building = { is_building_type = building_barracks occupancy >= 0.7 } }
 }
}
