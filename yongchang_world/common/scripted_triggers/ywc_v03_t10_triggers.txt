# JHG is the jurisdiction country (JE owner); DMG is the trade country.

# Entry: JHG holds a real staffed port and trade center, its customs courts
# have a real staffing gap for the claim to land in, and DMG has its own
# port or trade presence so the route is real on both ends.

 c:JHG ?= this
 any_scope_state = { owner = ROOT any_scope_building = { is_building_type = building_port occupancy >= 0.5 } }

  is_country_alive = yes
  has_diplomatic_relevance = c:JHG
  OR = {
   any_scope_state = { owner = c:DMG any_scope_building = { is_building_type = building_port occupancy >= 0.3 } }
   any_scope_state = { owner = c:DMG any_scope_building = { is_building_type = building_trade_center occupancy >= 0.3 } }
  }
 }
}
