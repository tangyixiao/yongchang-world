# Regional MHG (F63) support effects. The common-resident journal opens only
# from a real land-and-waterway choice, never at startup.
ywc_mhg_open_resident_je = {
	if = {
		limit = {
			NOT = { has_journal_entry = ywc_je_v03_mhg_common_resident }
			NOT = { has_variable = ywc_mhg_resident_started }
			is_country_alive = yes
		}
		add_journal_entry = { type = ywc_je_v03_mhg_common_resident }
	}
}
