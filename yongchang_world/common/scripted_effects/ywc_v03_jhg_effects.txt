ywc_jhg_court_close_as_established = {
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			NOT = { has_variable = ywc_jhg_court_established }
			NOT = { has_variable = ywc_jhg_court_withdrawn }
			has_variable = ywc_jhg_court_six_months_complete
			ywc_jhg_court_current_target_ready = yes
		}
		set_variable = { name = ywc_jhg_court_established value = 1 }
		set_variable = { name = ywc_jhg_v03_jhg_court_assembly_outcome value = 1 }
		ywc_jhg_court_finalize = yes
	}
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			NOT = { ywc_jhg_court_current_target_ready = yes }
		}
		ywc_jhg_court_invalidate_stale_watch = yes
	}
}

ywc_jhg_court_close_as_compromise = {
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			NOT = { has_variable = ywc_jhg_court_established }
			NOT = { has_variable = ywc_jhg_court_withdrawn }
			var:ywc_jhg_court_route > 0
			OR = { has_variable = ywc_jhg_court_six_months_complete has_variable = ywc_jhg_court_limited_six_months_complete }

		set_variable = { name = ywc_jhg_court_compromise value = 1 }
		set_variable = { name = ywc_jhg_v03_jhg_court_assembly_outcome value = 2 }
		ywc_jhg_court_finalize = yes
	}
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			NOT = { ywc_jhg_court_current_target_ready = yes }
		}
		ywc_jhg_court_invalidate_stale_watch = yes
	}
}

ywc_jhg_court_close_as_breach = {
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
		}
		set_variable = { name = ywc_jhg_court_withdrawn value = 1 }
		set_variable = { name = ywc_jhg_v03_jhg_court_assembly_outcome value = 4 }
		ywc_jhg_court_finalize = yes
	}
}

ywc_jhg_court_finalize = {
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			var:ywc_jhg_v03_jhg_court_assembly_outcome > 0
		}
		set_variable = { name = ywc_jhg_v03_jhg_court_assembly_settled value = 1 }
		change_variable = { name = ywc_v03_active_arc_count add = -1 }
		change_variable = { name = ywc_v03_visible_task_count add = -1 }
		clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 }
		clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 }
		ywc_jhg_court_reset_observation = yes
			remove_variable = ywc_jhg_court_notice_delivery_active
	remove_variable = ywc_jhg_court_notice_delivery_12
	remove_variable = ywc_jhg_court_notice_delivery_36
	remove_variable = ywc_jhg_court_notice_pending_event
		ywc_jhg_court_bridge_legacy = yes
	}
}

ywc_jhg_court_invalidate_stale_watch = {
	if = {
		limit = { NOT = { ywc_jhg_court_current_target_ready = yes } }
		ywc_jhg_court_reset_observation = yes
		remove_variable = ywc_jhg_court_six_months_complete
		remove_variable = ywc_jhg_court_limited_six_months_complete
	}
}

ywc_jhg_court_reset_observation = {
	remove_variable = ywc_jhg_court_observation_armed
	remove_variable = ywc_jhg_court_observation_timer
}

ywc_jhg_court_monthly_review = {
	if = {
		limit = { NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled } }
		if = {
			limit = { var:ywc_jhg_court_route > 0 var:ywc_jhg_court_observation_route != var:ywc_jhg_court_route }
			ywc_jhg_court_reset_observation = yes
			remove_variable = ywc_jhg_court_six_months_complete
			remove_variable = ywc_jhg_court_limited_six_months_complete
			set_variable = { name = ywc_jhg_court_observation_route value = var:ywc_jhg_court_route }
		}
		if = {
			limit = { ywc_jhg_court_current_target_ready = yes }
			ywc_jhg_finance_review = yes
			remove_variable = ywc_jhg_court_blocked_months
			remove_variable = ywc_jhg_court_12_notice
			remove_variable = ywc_jhg_court_36_notice
			remove_variable = ywc_jhg_court_notice_pending_event
			remove_variable = ywc_jhg_court_12_notice_delivered
			remove_variable = ywc_jhg_court_36_notice_delivered
			remove_variable = ywc_jhg_court_notice_delivery_active
			remove_variable = ywc_jhg_court_notice_delivery_12
			remove_variable = ywc_jhg_court_notice_delivery_36
			set_variable = { name = ywc_jhg_court_observation_route value = var:ywc_jhg_court_route }
			if = {
				limit = { NOT = { has_variable = ywc_jhg_court_observation_armed } }
				set_variable = { name = ywc_jhg_court_observation_armed value = 1 }
				set_variable = { name = ywc_jhg_court_observation_timer value = 0 days = 180 }
			}
			else_if = {
				limit = {
					has_variable = ywc_jhg_court_observation_armed
					NOT = { has_variable = ywc_jhg_court_observation_timer }
				}
				if = { limit = { has_variable = ywc_jhg_court_limited_target_selected } set_variable = { name = ywc_jhg_court_limited_six_months_complete value = 1 } set_variable = { name = ywc_jhg_court_notice_pending_event value = 1007 } } else = { set_variable = { name = ywc_jhg_court_six_months_complete value = 1 } }
			}
		}
		else = {
			ywc_jhg_court_reset_observation = yes
			remove_variable = ywc_jhg_court_six_months_complete
			remove_variable = ywc_jhg_court_limited_six_months_complete
			if = { limit = { NOT = { has_variable = ywc_jhg_court_blocked_months } } set_variable = { name = ywc_jhg_court_blocked_months value = 0 } }
			change_variable = { name = ywc_jhg_court_blocked_months add = 1 }
			if = {
				limit = { var:ywc_jhg_court_blocked_months >= 12 NOT = { has_variable = ywc_jhg_court_12_notice_delivered } }
				set_variable = { name = ywc_jhg_court_12_notice value = 1 days = 365 }
				set_variable = { name = ywc_jhg_court_notice_pending_event value = 1007 }
			}
			if = {
				limit = { var:ywc_jhg_court_blocked_months >= 36 NOT = { has_variable = ywc_jhg_court_36_notice_delivered } }
				set_variable = { name = ywc_jhg_court_36_notice value = 1 days = 1095 }
				set_variable = { name = ywc_jhg_court_notice_pending_event value = 1007 }
			}
		}
	}
}

ywc_jhg_court_bridge_legacy = {
	if = {
		limit = { NOT = { has_variable = ywc_campaign_jhg_c1_outcome } }
		set_variable = { name = ywc_campaign_jhg_c1_outcome value = var:ywc_jhg_v03_jhg_court_assembly_outcome }
	}
	else_if = {
		limit = { has_variable = ywc_campaign_jhg_c1_outcome NOT = { has_variable = ywc_jhg_court_prior_campaign_outcome } }
		set_variable = { name = ywc_jhg_court_prior_campaign_outcome value = var:ywc_campaign_jhg_c1_outcome }

			limit = { var:ywc_jhg_tribute_route > 0 var:ywc_jhg_tribute_observation_route != var:ywc_jhg_tribute_route }
			ywc_jhg_tribute_reset_observation = yes

			remove_variable = ywc_jhg_tribute_limited_six_months_complete
			set_variable = { name = ywc_jhg_tribute_observation_route value = var:ywc_jhg_tribute_route }
		}
		if = {
			limit = { ywc_jhg_tribute_current_target_ready = yes }
			remove_variable = ywc_jhg_tribute_blocked_months
			remove_variable = ywc_jhg_tribute_12_notice
			remove_variable = ywc_jhg_tribute_36_notice
			remove_variable = ywc_jhg_tribute_notice_pending_event
			remove_variable = ywc_jhg_tribute_12_notice_delivered
			remove_variable = ywc_jhg_tribute_36_notice_delivered
			remove_variable = ywc_jhg_tribute_notice_delivery_active
			remove_variable = ywc_jhg_tribute_notice_delivery_12
			remove_variable = ywc_jhg_tribute_notice_delivery_36
			set_variable = { name = ywc_jhg_tribute_observation_route value = var:ywc_jhg_tribute_route }
			if = {
				limit = { has_variable = ywc_jhg_tribute_settlement_stage }

					limit = { NOT = { has_variable = ywc_jhg_tribute_observation_armed } }
					set_variable = { name = ywc_jhg_tribute_observation_armed value = 1 }
					set_variable = { name = ywc_jhg_tribute_observation_timer value = 0 days = 180 }
				}
				else_if = {
					limit = {
						has_variable = ywc_jhg_tribute_observation_armed
						NOT = { has_variable = ywc_jhg_tribute_observation_timer }
					}
					if = { limit = { has_variable = ywc_jhg_tribute_limited_target_selected } set_variable = { name = ywc_jhg_tribute_limited_six_months_complete value = 1 } set_variable = { name = ywc_jhg_tribute_notice_pending_event value = 1025 } } else = { set_variable = { name = ywc_jhg_tribute_six_months_complete value = 1 } }
				}
			}
			else = {
				ywc_jhg_tribute_reset_observation = yes
				remove_variable = ywc_jhg_tribute_six_months_complete
				remove_variable = ywc_jhg_tribute_limited_six_months_complete
			}
		}
		else = {
			ywc_jhg_tribute_reset_observation = yes
			remove_variable = ywc_jhg_tribute_six_months_complete
			remove_variable = ywc_jhg_tribute_limited_six_months_complete
			if = { limit = { NOT = { has_variable = ywc_jhg_tribute_blocked_months } } set_variable = { name = ywc_jhg_tribute_blocked_months value = 0 } }
			change_variable = { name = ywc_jhg_tribute_blocked_months add = 1 }
			if = {
				limit = { var:ywc_jhg_tribute_blocked_months >= 12 NOT = { has_variable = ywc_jhg_tribute_12_notice_delivered } }
				set_variable = { name = ywc_jhg_tribute_12_notice value = 1 days = 365 }
				set_variable = { name = ywc_jhg_tribute_notice_pending_event value = 1025 }
			}
			if = {
				limit = { var:ywc_jhg_tribute_blocked_months >= 36 NOT = { has_variable = ywc_jhg_tribute_36_notice_delivered } }
				set_variable = { name = ywc_jhg_tribute_36_notice value = 1 days = 1095 }
				set_variable = { name = ywc_jhg_tribute_notice_pending_event value = 1025 }
			}
		}
	}
}

		if = {
			limit = {
				NOT = { has_variable = ywc_jhg_escort_term_timer }
				NOT = { var:ywc_jhg_escort_ended_generation ?= var:ywc_jhg_escort_contract_generation }
			}
			ywc_jhg_escort_bump_serial = yes
			set_variable = { name = ywc_jhg_escort_ended_generation value = var:ywc_jhg_escort_contract_generation }
		}
	}
	ywc_jhg_escort_invalidate_pending = yes
	if = {
		limit = { var:ywc_jhg_escort_state ?= 2 }
		ywc_jhg_escort_sign_contract = yes
	}
}

ywc_jhg_escort_end_partnership = {
	if = {
		limit = {
			exists = scope:ywc_jhg_escort_end_owner
			scope:ywc_jhg_escort_end_owner ?= this
			has_variable = ywc_jhg_escort_contract_generation
			var:ywc_jhg_escort_contract_generation ?= scope:ywc_jhg_escort_end_generation
			NOT = { var:ywc_jhg_escort_ended_generation ?= scope:ywc_jhg_escort_end_generation }
		}
		ywc_jhg_escort_bump_serial = yes
		set_variable = { name = ywc_jhg_escort_ended_generation value = scope:ywc_jhg_escort_end_generation }
		remove_variable = ywc_jhg_escort_term_timer
	}
}

# SHU and JHG each pay one captured share into the project budget. These are
# actual treasury debits, not a promise that a building was constructed.
ywc_jhg_share_bump_serial = {
	if = { limit = { NOT = { has_variable = ywc_jhg_share_serial } } set_variable = { name = ywc_jhg_share_serial value = 0 } }
	change_variable = { name = ywc_jhg_share_serial add = 1 }
}

ywc_jhg_share_send_shu = {
	if = {
		limit = { ywc_jhg_share_can_offer = yes }
		if = { limit = { NOT = { has_variable = ywc_jhg_share_generation } } set_variable = { name = ywc_jhg_share_generation value = 0 } }
		change_variable = { name = ywc_jhg_share_generation add = 1 }
		ywc_jhg_share_cancel_unsigned = yes
		ywc_jhg_share_bump_serial = yes
		c:SHU = { save_scope_as = ywc_jhg_share_request_partner }
		set_variable = { name = ywc_jhg_share_partner value = scope:ywc_jhg_share_request_partner }
		set_variable = { name = ywc_jhg_share_state value = 1 }
		set_variable = { name = ywc_jhg_share_request_timer value = 1 months = 3 }
		set_variable = { name = ywc_jhg_share_quote_generation value = var:ywc_jhg_share_generation }
		set_variable = { name = ywc_jhg_share_quote_serial value = var:ywc_jhg_share_serial }
		set_variable = { name = ywc_jhg_share_quote_jhg_cost value = var:ywc_jhg_tribute_filing_fee }
		set_variable = { name = ywc_jhg_share_quote_shu_cost value = var:ywc_jhg_tribute_filing_fee }
		set_variable = { name = ywc_jhg_share_quote_term value = 120 }
		save_scope_as = ywc_jhg_share_requester
		save_scope_value_as = { name = ywc_jhg_share_request_generation value = var:ywc_jhg_share_quote_generation }
		save_scope_value_as = { name = ywc_jhg_share_request_serial value = var:ywc_jhg_share_quote_serial }
		save_scope_value_as = { name = ywc_jhg_share_request_jhg_cost value = var:ywc_jhg_share_quote_jhg_cost }
		save_scope_value_as = { name = ywc_jhg_share_request_shu_cost value = var:ywc_jhg_share_quote_shu_cost }
		save_scope_value_as = { name = ywc_jhg_share_request_term value = var:ywc_jhg_share_quote_term }
		scope:ywc_jhg_share_request_partner = { trigger_event = { id = ywc_jhg.1026 } }
	}
}

		set_variable = { name = ywc_jhg_share_answer_generation value = scope:ywc_jhg_share_request_generation }
		set_variable = { name = ywc_jhg_share_answer_serial value = scope:ywc_jhg_share_request_serial }
		set_variable = { name = ywc_jhg_share_answer value = 1 }
		set_variable = { name = ywc_jhg_share_answer_jhg_cost value = scope:ywc_jhg_share_request_jhg_cost }
		set_variable = { name = ywc_jhg_share_answer_shu_cost value = scope:ywc_jhg_share_request_shu_cost }
		set_variable = { name = ywc_jhg_share_answer_term value = scope:ywc_jhg_share_request_term }
		scope:ywc_jhg_share_requester = {

			set_variable = { name = ywc_jhg_share_accept_serial value = scope:ywc_jhg_share_request_serial }
			set_variable = { name = ywc_jhg_share_accept_jhg_cost value = scope:ywc_jhg_share_request_jhg_cost }
			set_variable = { name = ywc_jhg_share_accept_shu_cost value = scope:ywc_jhg_share_request_shu_cost }

ywc_jhg_share_refuse = {

ywc_jhg_share_cancel_unsigned = {
	if = {
		limit = { OR = { var:ywc_jhg_share_state ?= 1 var:ywc_jhg_share_state ?= 2 } }
		remove_variable = ywc_jhg_share_request_timer
		remove_variable = ywc_jhg_share_accept_generation
		remove_variable = ywc_jhg_share_accept_serial
		remove_variable = ywc_jhg_share_accept_jhg_cost
		remove_variable = ywc_jhg_share_accept_shu_cost
		remove_variable = ywc_jhg_share_accept_term
		remove_variable = ywc_jhg_share_partner
		ywc_jhg_share_bump_serial = yes
		set_variable = { name = ywc_jhg_share_state value = 7 }
	}
}

ywc_jhg_share_sign_contract = {
	save_scope_as = ywc_jhg_share_payment_owner
	if = {
		limit = {
			ywc_jhg_share_accepted_current = yes
			ywc_jhg_finance_safe = yes
			gold_reserves >= var:ywc_jhg_share_quote_jhg_cost
			c:SHU = {
				net_fixed_income >= 0
				NOT = { in_default = yes }
				NOT = { has_modifier = declared_bankruptcy }
				gold_reserves >= scope:ywc_jhg_share_payment_owner.var:ywc_jhg_share_quote_shu_cost
			}
		}
		add_treasury = { value = var:ywc_jhg_share_quote_jhg_cost multiply = -1 }
		var:ywc_jhg_share_partner = { add_treasury = { value = scope:ywc_jhg_share_payment_owner.var:ywc_jhg_share_quote_shu_cost multiply = -1 } }
		set_variable = { name = ywc_jhg_share_contract_partner value = var:ywc_jhg_share_partner }
		set_variable = { name = ywc_jhg_share_contract_generation value = var:ywc_jhg_share_quote_generation }
		set_variable = { name = ywc_jhg_share_contract_offer_serial value = var:ywc_jhg_share_quote_serial }
		set_variable = { name = ywc_jhg_share_contract_jhg_cost value = var:ywc_jhg_share_quote_jhg_cost }
		set_variable = { name = ywc_jhg_share_contract_shu_cost value = var:ywc_jhg_share_quote_shu_cost }
		set_variable = { name = ywc_jhg_share_contract_term value = var:ywc_jhg_share_quote_term }
		set_variable = { name = ywc_jhg_share_current_month value = ywc_jhg_current_calendar_month }
		set_variable = { name = ywc_jhg_share_contract_end_month value = var:ywc_jhg_share_current_month }
		change_variable = { name = ywc_jhg_share_contract_end_month add = var:ywc_jhg_share_quote_term }
		set_variable = { name = ywc_jhg_share_term_timer value = 1 months = 120 }
		remove_variable = ywc_jhg_share_request_timer
		remove_variable = ywc_jhg_share_ended_generation
		remove_variable = ywc_jhg_share_suspended_generation
		ywc_jhg_share_bump_serial = yes
		set_variable = { name = ywc_jhg_share_state value = 6 }
		save_scope_as = ywc_jhg_share_end_owner
		save_scope_value_as = { name = ywc_jhg_share_end_generation value = var:ywc_jhg_share_contract_generation }
		trigger_event = { id = ywc_jhg.1027 months = 120 }
	}
}

ywc_jhg_share_monthly = {
	if = {
		limit = { OR = { var:ywc_jhg_share_state ?= 1 var:ywc_jhg_share_state ?= 2 } }
		if = {
			limit = { OR = { NOT = { has_variable = ywc_jhg_share_request_timer } NOT = { ywc_jhg_share_pending_partner_intact = yes } } }
			ywc_jhg_share_cancel_unsigned = yes
		}
		else_if = { limit = { var:ywc_jhg_share_state ?= 2 } ywc_jhg_share_sign_contract = yes }
	}
	if = {
		limit = { has_variable = ywc_jhg_share_contract_generation }
		set_variable = { name = ywc_jhg_share_current_month value = ywc_jhg_current_calendar_month }
		if = {
			limit = { NOT = { ywc_jhg_share_contract_partner_intact = yes } NOT = { var:ywc_jhg_share_suspended_generation ?= var:ywc_jhg_share_contract_generation } }
			ywc_jhg_share_bump_serial = yes
			set_variable = { name = ywc_jhg_share_suspended_generation value = var:ywc_jhg_share_contract_generation }
		}
		else_if = {
			limit = { ywc_jhg_share_contract_partner_intact = yes var:ywc_jhg_share_suspended_generation ?= var:ywc_jhg_share_contract_generation }
			remove_variable = ywc_jhg_share_suspended_generation
			ywc_jhg_share_bump_serial = yes
		}
		if = {
			limit = { NOT = { has_variable = ywc_jhg_share_term_timer } NOT = { var:ywc_jhg_share_ended_generation ?= var:ywc_jhg_share_contract_generation } }
			ywc_jhg_share_bump_serial = yes
			set_variable = { name = ywc_jhg_share_ended_generation value = var:ywc_jhg_share_contract_generation }
		}
	}
}

ywc_jhg_share_end_partnership = {
	if = {
		limit = { exists = scope:ywc_jhg_share_end_owner scope:ywc_jhg_share_end_owner ?= this has_variable = ywc_jhg_share_contract_generation var:ywc_jhg_share_contract_generation ?= scope:ywc_jhg_share_end_generation NOT = { var:ywc_jhg_share_ended_generation ?= scope:ywc_jhg_share_end_generation } }
		ywc_jhg_share_bump_serial = yes
		set_variable = { name = ywc_jhg_share_ended_generation value = scope:ywc_jhg_share_end_generation }
		remove_variable = ywc_jhg_share_term_timer
	}
}
