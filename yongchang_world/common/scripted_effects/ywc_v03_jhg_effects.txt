ywc_jhg_court_close_as_established = {
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			NOT = { has_variable = ywc_jhg_court_established }
			NOT = { has_variable = ywc_jhg_court_withdrawn }
			has_variable = ywc_jhg_court_six_months_complete
			ywc_jhg_court_current_target_ready = yes
		}
		set_variable = { name = ywc_jhg_court_established value = 1 }
		set_variable = { name = ywc_jhg_v03_jhg_court_assembly_outcome value = 1 }
		ywc_jhg_court_finalize = yes
	}
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			NOT = { ywc_jhg_court_current_target_ready = yes }
		}
		ywc_jhg_court_invalidate_stale_watch = yes
	}
}

ywc_jhg_court_close_as_compromise = {
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			NOT = { has_variable = ywc_jhg_court_established }
			NOT = { has_variable = ywc_jhg_court_withdrawn }
			var:ywc_jhg_court_route > 0
			OR = { has_variable = ywc_jhg_court_six_months_complete has_variable = ywc_jhg_court_limited_six_months_complete }
			ywc_jhg_court_current_target_ready = yes
		}
		set_variable = { name = ywc_jhg_court_compromise value = 1 }
		set_variable = { name = ywc_jhg_v03_jhg_court_assembly_outcome value = 2 }
		ywc_jhg_court_finalize = yes
	}
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			NOT = { ywc_jhg_court_current_target_ready = yes }
		}
		ywc_jhg_court_invalidate_stale_watch = yes
	}
}

ywc_jhg_court_close_as_breach = {
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
		}
		set_variable = { name = ywc_jhg_court_withdrawn value = 1 }
		set_variable = { name = ywc_jhg_v03_jhg_court_assembly_outcome value = 4 }
		ywc_jhg_court_finalize = yes
	}
}

ywc_jhg_court_finalize = {
	if = {
		limit = {
			has_variable = ywc_jhg_v03_jhg_court_assembly_started
			NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled }
			var:ywc_jhg_v03_jhg_court_assembly_outcome > 0
		}
		set_variable = { name = ywc_jhg_v03_jhg_court_assembly_settled value = 1 }
		change_variable = { name = ywc_v03_active_arc_count add = -1 }
		change_variable = { name = ywc_v03_visible_task_count add = -1 }
		clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 }
		clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 }
		ywc_jhg_court_reset_observation = yes
			remove_variable = ywc_jhg_court_notice_delivery_active
	remove_variable = ywc_jhg_court_notice_delivery_12
	remove_variable = ywc_jhg_court_notice_delivery_36
	remove_variable = ywc_jhg_court_notice_pending_event
		ywc_jhg_court_bridge_legacy = yes
	}
}

ywc_jhg_court_invalidate_stale_watch = {
	if = {
		limit = { NOT = { ywc_jhg_court_current_target_ready = yes } }
		ywc_jhg_court_reset_observation = yes
		remove_variable = ywc_jhg_court_six_months_complete
		remove_variable = ywc_jhg_court_limited_six_months_complete
	}
}

ywc_jhg_court_reset_observation = {
	remove_variable = ywc_jhg_court_observation_armed
	remove_variable = ywc_jhg_court_observation_timer
}

ywc_jhg_court_monthly_review = {
	if = {
		limit = { NOT = { has_variable = ywc_jhg_v03_jhg_court_assembly_settled } }
		if = {
			limit = { var:ywc_jhg_court_route > 0 var:ywc_jhg_court_observation_route != var:ywc_jhg_court_route }
			ywc_jhg_court_reset_observation = yes
			remove_variable = ywc_jhg_court_six_months_complete
			remove_variable = ywc_jhg_court_limited_six_months_complete
			set_variable = { name = ywc_jhg_court_observation_route value = var:ywc_jhg_court_route }
		}
		if = {
			limit = { ywc_jhg_court_current_target_ready = yes }
			ywc_jhg_finance_review = yes
			remove_variable = ywc_jhg_court_blocked_months
			remove_variable = ywc_jhg_court_12_notice
			remove_variable = ywc_jhg_court_36_notice
			remove_variable = ywc_jhg_court_notice_pending_event
			remove_variable = ywc_jhg_court_12_notice_delivered
			remove_variable = ywc_jhg_court_36_notice_delivered
			remove_variable = ywc_jhg_court_notice_delivery_active
			remove_variable = ywc_jhg_court_notice_delivery_12
			remove_variable = ywc_jhg_court_notice_delivery_36
			set_variable = { name = ywc_jhg_court_observation_route value = var:ywc_jhg_court_route }
			if = {
				limit = { NOT = { has_variable = ywc_jhg_court_observation_armed } }
				set_variable = { name = ywc_jhg_court_observation_armed value = 1 }
				set_variable = { name = ywc_jhg_court_observation_timer value = 0 days = 180 }
			}
			else_if = {
				limit = {
					has_variable = ywc_jhg_court_observation_armed
					NOT = { has_variable = ywc_jhg_court_observation_timer }
				}
				if = { limit = { has_variable = ywc_jhg_court_limited_target_selected } set_variable = { name = ywc_jhg_court_limited_six_months_complete value = 1 } set_variable = { name = ywc_jhg_court_notice_pending_event value = 1007 } } else = { set_variable = { name = ywc_jhg_court_six_months_complete value = 1 } }
			}
		}
		else = {
			ywc_jhg_court_reset_observation = yes
			remove_variable = ywc_jhg_court_six_months_complete
			remove_variable = ywc_jhg_court_limited_six_months_complete
			if = { limit = { NOT = { has_variable = ywc_jhg_court_blocked_months } } set_variable = { name = ywc_jhg_court_blocked_months value = 0 } }
			change_variable = { name = ywc_jhg_court_blocked_months add = 1 }
			if = {
				limit = { var:ywc_jhg_court_blocked_months >= 12 NOT = { has_variable = ywc_jhg_court_12_notice_delivered } }
				set_variable = { name = ywc_jhg_court_12_notice value = 1 days = 365 }
				set_variable = { name = ywc_jhg_court_notice_pending_event value = 1007 }
			}
			if = {
				limit = { var:ywc_jhg_court_blocked_months >= 36 NOT = { has_variable = ywc_jhg_court_36_notice_delivered } }
				set_variable = { name = ywc_jhg_court_36_notice value = 1 days = 1095 }
				set_variable = { name = ywc_jhg_court_notice_pending_event value = 1007 }
			}
		}
	}
}

ywc_jhg_court_bridge_legacy = {
	if = {
		limit = { NOT = { has_variable = ywc_campaign_jhg_c1_outcome } }
		set_variable = { name = ywc_campaign_jhg_c1_outcome value = var:ywc_jhg_v03_jhg_court_assembly_outcome }
	}
	else_if = {
		limit = { has_variable = ywc_campaign_jhg_c1_outcome NOT = { has_variable = ywc_jhg_court_prior_campaign_outcome } }
		set_variable = { name = ywc_jhg_court_prior_campaign_outcome value = var:ywc_campaign_jhg_c1_outcome }
	}
}

ywc_jhg_tribute_monthly_review = {

		if = {
			limit = { var:ywc_jhg_tribute_route > 0 var:ywc_jhg_tribute_observation_route != var:ywc_jhg_tribute_route }
			ywc_jhg_tribute_reset_observation = yes
			remove_variable = ywc_jhg_tribute_six_months_complete
			remove_variable = ywc_jhg_tribute_limited_six_months_complete
			set_variable = { name = ywc_jhg_tribute_observation_route value = var:ywc_jhg_tribute_route }
		}
		if = {
			limit = { ywc_jhg_tribute_current_target_ready = yes }
			remove_variable = ywc_jhg_tribute_blocked_months
			remove_variable = ywc_jhg_tribute_12_notice
			remove_variable = ywc_jhg_tribute_36_notice
			remove_variable = ywc_jhg_tribute_notice_pending_event
			remove_variable = ywc_jhg_tribute_12_notice_delivered
			remove_variable = ywc_jhg_tribute_36_notice_delivered
			remove_variable = ywc_jhg_tribute_notice_delivery_active
			remove_variable = ywc_jhg_tribute_notice_delivery_12
			remove_variable = ywc_jhg_tribute_notice_delivery_36
			set_variable = { name = ywc_jhg_tribute_observation_route value = var:ywc_jhg_tribute_route }
			if = {
				limit = { has_variable = ywc_jhg_tribute_settlement_stage }
				if = {
					limit = { NOT = { has_variable = ywc_jhg_tribute_observation_armed } }
					set_variable = { name = ywc_jhg_tribute_observation_armed value = 1 }
					set_variable = { name = ywc_jhg_tribute_observation_timer value = 0 days = 180 }
				}
				else_if = {
					limit = {
						has_variable = ywc_jhg_tribute_observation_armed
						NOT = { has_variable = ywc_jhg_tribute_observation_timer }
					}
					if = { limit = { has_variable = ywc_jhg_tribute_limited_target_selected } set_variable = { name = ywc_jhg_tribute_limited_six_months_complete value = 1 } set_variable = { name = ywc_jhg_tribute_notice_pending_event value = 1025 } } else = { set_variable = { name = ywc_jhg_tribute_six_months_complete value = 1 } }
				}
			}
			else = {
				ywc_jhg_tribute_reset_observation = yes
				remove_variable = ywc_jhg_tribute_six_months_complete
				remove_variable = ywc_jhg_tribute_limited_six_months_complete
			}
		}
		else = {
			ywc_jhg_tribute_reset_observation = yes
			remove_variable = ywc_jhg_tribute_six_months_complete
			remove_variable = ywc_jhg_tribute_limited_six_months_complete
			if = { limit = { NOT = { has_variable = ywc_jhg_tribute_blocked_months } } set_variable = { name = ywc_jhg_tribute_blocked_months value = 0 } }
			change_variable = { name = ywc_jhg_tribute_blocked_months add = 1 }
			if = {
				limit = { var:ywc_jhg_tribute_blocked_months >= 12 NOT = { has_variable = ywc_jhg_tribute_12_notice_delivered } }
				set_variable = { name = ywc_jhg_tribute_12_notice value = 1 days = 365 }
				set_variable = { name = ywc_jhg_tribute_notice_pending_event value = 1025 }
			}
			if = {
				limit = { var:ywc_jhg_tribute_blocked_months >= 36 NOT = { has_variable = ywc_jhg_tribute_36_notice_delivered } }
				set_variable = { name = ywc_jhg_tribute_36_notice value = 1 days = 1095 }
				set_variable = { name = ywc_jhg_tribute_notice_pending_event value = 1025 }
			}
		}
	}
}
