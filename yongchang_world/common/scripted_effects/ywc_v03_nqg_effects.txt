	else_if = { limit = { ywc_nqg_aid_can_offer_rus = yes } c:RUS = { save_scope_as = ywc_nqg_aid_request_partner } ywc_nqg_aid_send_captured_inquiry = yes }

ywc_nqg_aid_send_captured_inquiry = {
	if = {
		limit = {
			exists = scope:ywc_nqg_aid_request_partner
			OR = {
				AND = { scope:ywc_nqg_aid_request_partner ?= c:SHU ywc_nqg_aid_can_offer_shu = yes }
				AND = { scope:ywc_nqg_aid_request_partner ?= c:RUS ywc_nqg_aid_can_offer_rus = yes }
				AND = { scope:ywc_nqg_aid_request_partner ?= c:JAP ywc_nqg_aid_can_offer_jap = yes }
			}
		}
		if = { limit = { NOT = { has_variable = ywc_nqg_aid_generation } } set_variable = { name = ywc_nqg_aid_generation value = 0 } }
		change_variable = { name = ywc_nqg_aid_generation add = 1 }
		ywc_nqg_aid_cancel_unsigned = yes
		set_variable = { name = ywc_nqg_aid_partner value = scope:ywc_nqg_aid_request_partner }
		set_variable = { name = ywc_nqg_aid_state value = 1 }
		set_variable = { name = ywc_nqg_aid_request_timer value = 1 months = 3 }
		set_variable = { name = ywc_nqg_aid_total value = var:ywc_nqg_winter_fee }
		change_variable = { name = ywc_nqg_aid_total multiply = 3 }
		remove_variable = ywc_nqg_winter_aid_notice_open_serial
		remove_variable = ywc_nqg_winter_aid_notice_event
		save_scope_as = ywc_nqg_aid_requester
		save_scope_value_as = { name = ywc_nqg_aid_request_generation value = var:ywc_nqg_aid_generation }
		save_scope_value_as = { name = ywc_nqg_aid_request_total value = var:ywc_nqg_aid_total }
		scope:ywc_nqg_aid_request_partner = { trigger_event = { id = ywc_nqg.1030 } }
	}
}

				owner ?= c:NQG

		set_variable = { name = ywc_nqg_aid_paid_total value = 0 }
		set_variable = { name = ywc_nqg_aid_earned_count value = 0 }
		set_variable = { name = ywc_nqg_aid_earned_total value = 0 }
		remove_variable = ywc_nqg_aid_request_timer
		remove_variable = ywc_nqg_aid_quote_generation
		remove_variable = ywc_nqg_aid_quote_total
		remove_variable = ywc_nqg_aid_window_start_month
		remove_variable = ywc_nqg_aid_window_active

		remove_variable = ywc_nqg_aid_outcome_generation
		remove_variable = ywc_nqg_aid_creditor_invalid_generation

# One owner: the NQG country-monthly on_action. Its calendar token is separate
# from the winter arc's own six-month observation timer.
ywc_nqg_aid_monthly = {
	if = {
		limit = { has_variable = ywc_nqg_aid_contract_generation }
		save_scope_as = ywc_nqg_aid_requester
		if = { limit = { exists = var:ywc_nqg_aid_contract_partner } var:ywc_nqg_aid_contract_partner = { save_scope_as = ywc_nqg_aid_provider } }
		if = { limit = { exists = var:ywc_nqg_aid_source_state } save_scope_value_as = { name = ywc_nqg_aid_source_state value = var:ywc_nqg_aid_source_state } }
		if = { limit = { exists = var:ywc_nqg_aid_dest_state } save_scope_value_as = { name = ywc_nqg_aid_dest_state value = var:ywc_nqg_aid_dest_state } }
		if = { limit = { exists = var:ywc_nqg_aid_source_market } save_scope_value_as = { name = ywc_nqg_aid_source_market value = var:ywc_nqg_aid_source_market } }
		if = { limit = { exists = var:ywc_nqg_aid_dest_market } save_scope_value_as = { name = ywc_nqg_aid_dest_market value = var:ywc_nqg_aid_dest_market } }
		set_variable = { name = ywc_nqg_aid_current_month value = ywc_nqg_current_calendar_month }
		if = {
			limit = { NOT = { var:ywc_nqg_aid_last_processed_calendar_month ?= var:ywc_nqg_aid_current_month } }
			set_variable = { name = ywc_nqg_aid_last_processed_calendar_month value = var:ywc_nqg_aid_current_month }
			if = {
				limit = {
					has_variable = ywc_nqg_aid_window_active
					NOT = { has_variable = ywc_v03_nqg_v03_nqg_winter_court_settled }
					var:ywc_nqg_aid_current_month < { value = var:ywc_nqg_aid_window_start_month add = 3 }
					ywc_nqg_aid_inspection_ready = yes
				}
				set_variable = { name = ywc_nqg_aid_earned_installment value = var:ywc_nqg_aid_contract_total }
				change_variable = { name = ywc_nqg_aid_earned_installment divide = 3 }
				if = {
					limit = { var:ywc_nqg_aid_earned_count >= 2 }
					set_variable = { name = ywc_nqg_aid_earned_installment value = var:ywc_nqg_aid_contract_total }
					change_variable = { name = ywc_nqg_aid_earned_installment add = { value = var:ywc_nqg_aid_earned_total multiply = -1 } }
				}
				if = {
					limit = { var:ywc_nqg_aid_earned_installment > 0 }
					change_variable = { name = ywc_nqg_aid_earned_count add = 1 }
					change_variable = { name = ywc_nqg_aid_earned_total add = var:ywc_nqg_aid_earned_installment }
				}
			}
			if = {
				limit = { NOT = { exists = var:ywc_nqg_aid_contract_partner } }
				set_variable = { name = ywc_nqg_aid_creditor_invalid_generation value = var:ywc_nqg_aid_contract_generation }
			}
			if = {
				limit = { exists = var:ywc_nqg_aid_contract_partner var:ywc_nqg_aid_contract_partner = { is_country_alive = no } }
				set_variable = { name = ywc_nqg_aid_creditor_invalid_generation value = var:ywc_nqg_aid_contract_generation }
			}
			ywc_nqg_aid_settle_installment = yes
			if = {
				limit = {
					has_variable = ywc_nqg_aid_window_start_month
					NOT = { var:ywc_nqg_aid_window_ended_generation ?= var:ywc_nqg_aid_contract_generation }
					OR = {
						NOT = { has_variable = ywc_nqg_aid_window_active }
						var:ywc_nqg_aid_current_month >= { value = var:ywc_nqg_aid_window_start_month add = 3 }
					}
				}
				remove_variable = ywc_nqg_aid_window_active
				set_variable = { name = ywc_nqg_aid_window_ended_generation value = var:ywc_nqg_aid_contract_generation }
			}
			if = {
				limit = {
					var:ywc_nqg_aid_window_ended_generation ?= var:ywc_nqg_aid_contract_generation
					NOT = { var:ywc_nqg_aid_outcome_generation ?= var:ywc_nqg_aid_contract_generation }
				}
				ywc_nqg_aid_evaluate_outcome = yes
			}
			ywc_nqg_aid_refresh_status = yes
		}
	}
	if = {
		limit = {
			OR = {
				var:ywc_nqg_aid_state ?= 1
				var:ywc_nqg_aid_state ?= 2
			}
		}
		ywc_nqg_aid_invalidate_pending = yes
	}
}

ywc_nqg_aid_settle_installment = {
	if = {
		limit = { ywc_nqg_aid_contract_collectible = yes }
		set_variable = { name = ywc_nqg_aid_transfer_amount value = var:ywc_nqg_aid_earned_total }
		change_variable = { name = ywc_nqg_aid_transfer_amount add = { value = var:ywc_nqg_aid_paid_total multiply = -1 } }
		set_variable = { name = ywc_nqg_aid_installment_basis value = var:ywc_nqg_aid_contract_total }
		change_variable = { name = ywc_nqg_aid_installment_basis divide = 3 }
		if = {
			limit = { var:ywc_nqg_aid_transfer_amount > var:ywc_nqg_aid_installment_basis }
			set_variable = { name = ywc_nqg_aid_transfer_amount value = var:ywc_nqg_aid_installment_basis }
		}
		if = {
			limit = {
				var:ywc_nqg_aid_transfer_amount > 0
				ywc_nqg_finance_safe = yes
				gold_reserves >= var:ywc_nqg_aid_transfer_amount
			}
			add_treasury = { value = var:ywc_nqg_aid_transfer_amount multiply = -1 }

					NOT = { c:NQG.market ?= var:ywc_nqg_port_treaty_dest_market }

				limit = { NOT = { var:ywc_nqg_port_treaty_lease_suspended_generation ?= var:ywc_nqg_port_treaty_contract_generation } }
				set_variable = { name = ywc_nqg_port_treaty_lease_suspended_generation value = var:ywc_nqg_port_treaty_contract_generation }

			}
			remove_variable = ywc_nqg_port_treaty_lease_suspended_generation
		}
		if = {
			limit = {
				has_variable = ywc_nqg_port_treaty_contract_end_month
				var:ywc_nqg_port_treaty_current_month >= var:ywc_nqg_port_treaty_contract_end_month

			}
			set_variable = { name = ywc_nqg_port_treaty_lease_ended_generation value = var:ywc_nqg_port_treaty_contract_generation }
		}
	}
