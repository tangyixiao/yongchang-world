# v0.3 SHU fiscal register. Long-lived variables represent actual public commitments,
# not a substitute for laws, employment, supply, treasury or diplomatic state.

ywc_shu_open_register_once = {
	if = {
		limit = { NOT = { has_variable = ywc_shu_register_open } }
		set_variable = { name = ywc_shu_register_open value = 1 }
		ywc_shu_capture_cost_base = yes
		set_variable = { name = ywc_shu_pilot_state_lost value = 0 }
		set_variable = { name = ywc_v03_shu_v03_shu_fiscal_register_started value = 1 }
	}
}

# Capture yearly gross income at activation and estimate one week as annual income / 52.
# Floor that amount at 100; the one-week audit is paid in two half-installments.
ywc_shu_capture_cost_base = {
	if = {
		limit = { NOT = { has_variable = ywc_shu_fiscal_cost_captured } }
		set_variable = { name = ywc_shu_fiscal_cost value = yearly_gross_income }
		change_variable = { name = ywc_shu_fiscal_cost divide = 52 }
		clamp_variable = { name = ywc_shu_fiscal_cost min = 100 max = 1000000000 }
		set_variable = { name = ywc_shu_fiscal_cost_captured value = 1 }
	}
}

ywc_shu_pay_audit_half_once = {
	if = {
		limit = {
			net_fixed_income >= 0
			NOT = { has_variable = ywc_shu_audit_installment_paid }
		}
		ywc_shu_capture_cost_base = yes
		if = {
			limit = { gold_reserves >= { value = var:ywc_shu_fiscal_cost multiply = 0.5 } }
			add_treasury = { value = var:ywc_shu_fiscal_cost multiply = -0.5 }
			set_variable = { name = ywc_shu_audit_installment_paid value = 1 }
		}
	}
}

ywc_shu_pay_audit_second_half_once = {
	if = {
		limit = {
			net_fixed_income >= 0
			NOT = { has_variable = ywc_shu_audit_installment_paid_second }
			gold_reserves >= { value = var:ywc_shu_fiscal_cost multiply = 0.5 }
		}
		add_treasury = { value = var:ywc_shu_fiscal_cost multiply = -0.5 }
		set_variable = { name = ywc_shu_audit_installment_paid_second value = 1 }
	}
}

ywc_shu_pay_compensation_once = {
	if = {
		limit = {
			net_fixed_income >= 0
			NOT = { has_variable = ywc_shu_compensation_paid }
		}
		ywc_shu_capture_cost_base = yes
		if = {
			limit = { gold_reserves >= { value = var:ywc_shu_fiscal_cost multiply = 2 } }
			add_treasury = { value = var:ywc_shu_fiscal_cost multiply = -2 }
			set_variable = { name = ywc_shu_compensation_paid value = 1 }
		}
	}
}

ywc_shu_mark_first_audit = {
	if = {
		limit = {
			has_variable = ywc_shu_audit_mode
			NOT = { has_variable = ywc_shu_first_audit }
			OR = {
				AND = { var:ywc_shu_audit_mode = 1 has_variable = ywc_shu_audit_installment_paid }
				AND = { var:ywc_shu_audit_mode = 2 has_variable = ywc_shu_local_audit_right }
				var:ywc_shu_audit_mode = 3
			}
			OR = {
				var:ywc_shu_audit_mode = 3
				AND = { var:ywc_shu_audit_mode = 2 has_variable = ywc_shu_local_audit_right ywc_shu_v03_has_pilot_state = yes }
				AND = { var:ywc_shu_audit_mode = 1 ywc_shu_v03_real_execution_ready = yes }
			}
		}
		set_variable = { name = ywc_shu_first_audit value = 1 }
		set_variable = { name = ywc_shu_second_audit_due days = 90 }
	}
}

# The country callback owns this ledger even after the fiscal arc exits.
ywc_shu_track_military_contract = {
 if = {
  limit = { has_variable = ywc_shu_military_contract_active ywc_shu_military_contract_valid = yes NOT = { has_variable = ywc_shu_military_next_installment } }
  if = { limit = { var:ywc_shu_military_installments_due < 12 } change_variable = { name = ywc_shu_military_installments_due add = 1 } }
  set_variable = { name = ywc_shu_military_next_installment value = 1 months = 1 }
  if = {
   limit = {
    var:ywc_shu_military_installments_paid < var:ywc_shu_military_installments_due
    NOT = { in_default = yes }
    NOT = { has_modifier = declared_bankruptcy }
    gold_reserves >= var:ywc_shu_military_installment
   }
   add_treasury = { value = var:ywc_shu_military_installment multiply = -1 }
   change_variable = { name = ywc_shu_military_installments_paid add = 1 }
  }
 }
 if = {
  limit = { has_variable = ywc_shu_military_contract_active ywc_shu_military_contract_valid = yes var:ywc_shu_military_installments_due >= 12 var:ywc_shu_military_installments_paid >= 12 NOT = { has_variable = ywc_shu_military_term_due } }
  remove_variable = ywc_shu_military_contract_active
  remove_variable = ywc_shu_military_next_installment
  set_variable = { name = ywc_shu_military_contract_fulfilled value = 1 }
 }
}

ywc_shu_accept_merchant_advance = {
	if = {
		limit = { ywc_shu_merchant_offer_ready = yes }
		ywc_shu_capture_cost_base = yes
		set_variable = { name = ywc_shu_merchant_principal value = var:ywc_shu_fiscal_cost }
		add_treasury = { value = var:ywc_shu_merchant_principal }
		set_variable = { name = ywc_shu_merchant_advance value = 1 }
		set_variable = { name = ywc_shu_merchant_advance_unpaid value = 1 }
		set_variable = { name = ywc_shu_merchant_repayment_due value = 1 months = 12 }
		set_variable = { name = ywc_shu_merchant_repayment_months value = 0 }
		set_variable = { name = ywc_shu_merchant_consultation value = 1 }
	}
}

ywc_shu_collect_merchant_repayment = {
	if = {
		limit = { has_variable = ywc_shu_merchant_advance_unpaid }
		change_variable = { name = ywc_shu_merchant_repayment_months add = 1 }
	}
	if = {
		limit = {
			has_variable = ywc_shu_merchant_advance_unpaid
			var:ywc_shu_merchant_repayment_months >= 12
			NOT = { has_variable = ywc_shu_merchant_repayment_due }
			gold_reserves >= { value = var:ywc_shu_merchant_principal }
		}
		add_treasury = { value = var:ywc_shu_merchant_principal multiply = -1 }
		remove_variable = ywc_shu_merchant_advance_unpaid
		remove_variable = ywc_shu_merchant_repayment_months
		set_variable = { name = ywc_shu_merchant_advance_repaid value = 1 }
		if = { limit = { NOT = { AND = { has_variable = ywc_shu_port_advance_unpaid OR = { NOT = { has_variable = ywc_shu_port_repayment_clock_initialized } has_variable = ywc_shu_port_legacy_shared_arrears } } } } remove_modifier = ywc_shu_local_credit_arrears }
	}
	else = {
		if = {
			limit = { has_variable = ywc_shu_merchant_advance_unpaid var:ywc_shu_merchant_repayment_months >= 12 NOT = { has_variable = ywc_shu_merchant_repayment_due } }
			add_modifier = { name = ywc_shu_local_credit_arrears months = 12 }
		}
	}
}

ywc_shu_record_second_audit = {
	if = {
		limit = {
			has_variable = ywc_shu_first_audit
			NOT = { has_variable = ywc_shu_second_audit }
			NOT = { has_variable = ywc_shu_second_audit_due }
			net_fixed_income >= 0
			ywc_shu_v03_real_execution_ready = yes
			OR = {
				AND = { var:ywc_shu_audit_mode = 1 has_variable = ywc_shu_audit_installment_paid var:ywc_shu_appeal_owner = 1 }
				AND = { var:ywc_shu_audit_mode = 2 has_variable = ywc_shu_local_audit_right var:ywc_shu_appeal_owner = 2 AND = { NOT = { has_variable = ywc_shu_merchant_advance_unpaid } OR = { AND = { has_variable = ywc_shu_merchant_consultation has_variable = ywc_shu_merchant_advance_repaid } AND = { has_variable = ywc_shu_merchant_advance_refused NOT = { has_variable = ywc_shu_merchant_advance } } } } }
				var:ywc_shu_audit_mode = 3
			}
		}
		if = {
			limit = { var:ywc_shu_audit_mode = 1 }
			ywc_shu_pay_audit_second_half_once = yes
		}
		if = {
			limit = { OR = { var:ywc_shu_audit_mode = 2 var:ywc_shu_audit_mode = 3 } }
			set_variable = { name = ywc_shu_audit_installment_paid_second value = 1 }
		}
		if = {
			limit = { has_variable = ywc_shu_audit_installment_paid_second }
			set_variable = { name = ywc_shu_second_audit value = 1 }
			set_variable = { name = ywc_shu_register_pilot_resolved value = 1 }
		}
	}
}

ywc_shu_track_debt_trend = {
	if = {
		limit = { NOT = { has_variable = ywc_shu_scaled_debt_previous } }
		set_variable = { name = ywc_shu_scaled_debt_previous value = scaled_debt }
		set_variable = { name = ywc_shu_scaled_debt_rise_months value = 0 }
	}
	else = {
		if = {
			limit = { scaled_debt > var:ywc_shu_scaled_debt_previous }

		}
		else = { set_variable = { name = ywc_shu_scaled_debt_rise_months value = 0 } }

	}
}

ywc_shu_resume_after_pilot_reselection = {
 if = { limit = { has_variable = ywc_shu_resume_event } set_variable = { name = ywc_shu_pending_event value = var:ywc_shu_resume_event } remove_variable = ywc_shu_resume_event }
 else_if = { limit = { var:ywc_v03_shu_v03_shu_fiscal_register_last_event = 1 } set_variable = { name = ywc_shu_pending_event value = 1001 } }
 else_if = { limit = { var:ywc_v03_shu_v03_shu_fiscal_register_last_event = 2 } set_variable = { name = ywc_shu_pending_event value = 1002 } }
 else_if = { limit = { var:ywc_v03_shu_v03_shu_fiscal_register_last_event = 3 } set_variable = { name = ywc_shu_pending_event value = 1003 } }

		# A real, still-present tax-book consultation right waives the same
		# consultation fee without any treasury transfer. An already paid fee is
		# never retroactively rewritten into a waiver and never refunded.
		if = {
			limit = { ywc_shu_merchant_consultation_source = yes }
			set_variable = { name = ywc_shu_port_fee_waived value = 1 }
		}
		else_if = {
			# The 1019 promise: a settled central register result plus the still-
			# owned original pilot state halves this verification fee. A leftover
			# consultation half-mode does not stack the two discounts into a
			# quarter fee; the halved fee stays a real treasury payment.
			limit = { ywc_shu_port_central_discount_ready = yes gold_reserves >= { value = var:ywc_shu_port_cost multiply = 0.5 } }
			set_variable = { name = ywc_shu_port_fee_amount value = var:ywc_shu_port_cost }
			change_variable = { name = ywc_shu_port_fee_amount multiply = 0.5 }
			set_variable = { name = ywc_shu_port_fee_paid value = 1 }
			add_treasury = { value = var:ywc_shu_port_fee_amount multiply = -1 }
		}
		else_if = {
			limit = { has_variable = ywc_shu_port_fee_half_mode gold_reserves >= { value = var:ywc_shu_port_cost multiply = 0.5 } }
			set_variable = { name = ywc_shu_port_fee_amount value = var:ywc_shu_port_cost }
			change_variable = { name = ywc_shu_port_fee_amount multiply = 0.5 }
			set_variable = { name = ywc_shu_port_fee_paid value = 1 }
			add_treasury = { value = var:ywc_shu_port_fee_amount multiply = -1 }
		}
		else_if = {
			limit = { gold_reserves >= { value = var:ywc_shu_port_cost } }
			set_variable = { name = ywc_shu_port_fee_amount value = var:ywc_shu_port_cost }
			set_variable = { name = ywc_shu_port_fee_paid value = 1 }

	}

# Six real calendar months mature; five is never six. Each pulse runs the
# current-condition interruption check first, so even a repeated same-month
# callback can never bypass a real current loss. The counting branch then adds
# at most one observation per DISTINCT global calendar month: the month read
# from the existing shared scalar ywc_nmg_current_calendar_month must differ

# branch has no NOT-paid gate, so a paid recount or a finite re-sign still

# Real emergency funding: drawn only if the reserve actually holds it, and the
# real debt book is read at the same moment.
ywc_shu_resil_fund_emergency_once = {
if = { limit = { ywc_shu_v03_resil_open = yes NOT = { has_variable = ywc_shu_resil_funded_used } }
 set_variable = { name = ywc_shu_resil_funded_used value = 1 days = 1 }
 if = { limit = { gold_reserves >= { value = var:ywc_shu_resil_cost multiply = 0.5 } }
  add_treasury = { value = var:ywc_shu_resil_cost multiply = -0.5 }
  set_variable = { name = ywc_shu_resil_funded value = 1 }
 }
 if = { limit = { in_default = yes } set_variable = { name = ywc_shu_resil_default_seen value = 1 } }
 remove_variable = ywc_shu_resil_funded_used
}
}

# Four real calendar months of standing readiness.
ywc_shu_resil_start_review = {
if = { limit = { ywc_shu_v03_resil_open = yes }
 set_variable = { name = ywc_shu_resil_review_running value = 1 }
 set_variable = { name = ywc_shu_resil_review_streak value = 0 }
 set_variable = { name = ywc_shu_resil_review_ticked value = 1 days = 1 }
 set_variable = { name = ywc_shu_resil_review_month value = ywc_nmg_current_calendar_month }
 remove_variable = ywc_shu_resil_review_mature
 remove_variable = ywc_shu_resil_review_interrupted
}
}

ywc_shu_resil_track_review = {
if = { limit = { ywc_shu_v03_resil_open = yes var:ywc_shu_resil_stage = 1223 has_variable = ywc_shu_resil_review_running }
 if = { limit = { ywc_shu_v03_resil_current_broken = yes }
  set_variable = { name = ywc_shu_resil_review_streak value = 0 }
  remove_variable = ywc_shu_resil_review_mature
  remove_variable = ywc_shu_resil_review_running
  set_variable = { name = ywc_shu_resil_review_interrupted value = 1 }
 }
 else_if = { limit = { NOT = { has_variable = ywc_shu_resil_review_ticked } }
  set_variable = { name = ywc_shu_resil_review_ticked value = 1 days = 1 }
  if = { limit = { NOT = { has_variable = ywc_shu_resil_review_month } }
   set_variable = { name = ywc_shu_resil_review_month value = ywc_nmg_current_calendar_month }
  }
  if = { limit = { NOT = { has_variable = ywc_shu_resil_review_mature } NOT = { var:ywc_shu_resil_review_month = ywc_nmg_current_calendar_month } }
   set_variable = { name = ywc_shu_resil_review_month value = ywc_nmg_current_calendar_month }
   change_variable = { name = ywc_shu_resil_review_streak add = 1 }
   if = { limit = { var:ywc_shu_resil_review_streak >= 4 }
    set_variable = { name = ywc_shu_resil_review_mature value = 1 }
    remove_variable = ywc_shu_resil_review_running
   }
  }
 }
}
}
