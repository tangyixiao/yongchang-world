ywc_nmg_release_slot = {
	change_variable = { name = ywc_v03_active_arc_count add = -1 }
	change_variable = { name = ywc_v03_visible_task_count add = -1 }
	clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 }
	clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 }
}

ywc_nmg_land_fee_snapshot = {
	if = {
		limit = { NOT = { has_variable = ywc_nmg_land_fee } }
		set_variable = { name = ywc_nmg_land_fee value = yearly_gross_income }
		change_variable = { name = ywc_nmg_land_fee divide = 52 }
		clamp_variable = { name = ywc_nmg_land_fee min = 100 max = 1000000000 }
	}
}

ywc_nmg_land_pay_fee_once = {
if = { limit = { NOT = { has_variable = ywc_nmg_land_fee } } ywc_nmg_land_fee_snapshot = yes }
if = {
  limit = { NOT = { has_variable = ywc_nmg_land_fee_paid } ywc_nmg_v03_finance_safe = yes gold_reserves >= var:ywc_nmg_land_fee }
  add_treasury = { value = var:ywc_nmg_land_fee multiply = -1 }
  set_variable = { name = ywc_nmg_land_fee_paid value = 1 }
}
}


ywc_nmg_claim_land_slot = {
	if = {
		limit = {
			NOT = { has_variable = ywc_v03_nmg_v03_nmg_land_arbitration_started }
			var:ywc_v03_active_arc_count < 3
			var:ywc_v03_visible_task_count < 6
		}
		ywc_v03_initialize_country = yes
		if = {
			limit = {
				NOT = { has_variable = ywc_v03_nmg_v03_nmg_land_arbitration_started }
				var:ywc_v03_active_arc_count < 3
				var:ywc_v03_visible_task_count < 6
			}
			set_variable = { name = ywc_v03_nmg_v03_nmg_land_arbitration_started value = 1 }
			set_variable = { name = ywc_v03_nmg_v03_nmg_land_arbitration_last_event value = 0 }
			set_variable = { name = ywc_v03_nmg_v03_nmg_land_arbitration_outcome value = 0 }
			change_variable = { name = ywc_v03_active_arc_count add = 1 }
			change_variable = { name = ywc_v03_visible_task_count add = 1 }
		}
	}
}

ywc_nmg_mexshare_monthly = {
	if = {
		limit = { has_variable = ywc_nmg_mexshare_contract_generation }
		save_scope_as = ywc_nmg_mexshare_payment_owner
		set_variable = { name = ywc_nmg_mexshare_current_month value = ywc_nmg_current_calendar_month }
		if = {
			limit = { NOT = { var:ywc_nmg_mexshare_last_month ?= var:ywc_nmg_mexshare_current_month } }
			set_variable = { name = ywc_nmg_mexshare_last_month value = var:ywc_nmg_mexshare_current_month }
			if = {
				limit = { OR = { NOT = { exists = var:ywc_nmg_mexshare_contract_partner } var:ywc_nmg_mexshare_contract_partner = { is_country_alive = no } } }
				set_variable = { name = ywc_nmg_mexshare_creditor_invalid_generation value = var:ywc_nmg_mexshare_contract_generation }
			}
			if = {
				limit = { NOT = { var:ywc_nmg_mexshare_ended_generation ?= var:ywc_nmg_mexshare_contract_generation } NOT = { var:ywc_nmg_mexshare_creditor_invalid_generation ?= var:ywc_nmg_mexshare_contract_generation } var:ywc_nmg_mexshare_earned_count < 60 var:ywc_nmg_mexshare_current_month <= { value = var:ywc_nmg_mexshare_start_month add = 60 } }
				change_variable = { name = ywc_nmg_mexshare_earned_total add = var:ywc_nmg_mexshare_contract_monthly }
				change_variable = { name = ywc_nmg_mexshare_earned_count add = 1 }
			}
			if = {
				limit = { var:ywc_nmg_mexshare_earned_total > var:ywc_nmg_mexshare_paid_total NOT = { var:ywc_nmg_mexshare_creditor_invalid_generation ?= var:ywc_nmg_mexshare_contract_generation } exists = var:ywc_nmg_mexshare_contract_partner var:ywc_nmg_mexshare_contract_partner = { is_country_alive = yes } ywc_nmg_v03_finance_safe = yes }
				# Pay all due amounts if affordable; otherwise pay one instalment
				# when possible so a temporary default can actually be cured.
				set_variable = { name = ywc_nmg_mexshare_payment_due value = var:ywc_nmg_mexshare_earned_total }
				change_variable = { name = ywc_nmg_mexshare_payment_due add = { value = var:ywc_nmg_mexshare_paid_total multiply = -1 } }
				if = { limit = { gold_reserves < var:ywc_nmg_mexshare_payment_due } set_variable = { name = ywc_nmg_mexshare_payment_due value = var:ywc_nmg_mexshare_contract_monthly } }
				if = {
					limit = { gold_reserves >= var:ywc_nmg_mexshare_payment_due }
					add_treasury = { value = var:ywc_nmg_mexshare_payment_due multiply = -1 }
					var:ywc_nmg_mexshare_contract_partner = { add_treasury = { value = scope:ywc_nmg_mexshare_payment_owner.var:ywc_nmg_mexshare_payment_due } }
					change_variable = { name = ywc_nmg_mexshare_paid_total add = var:ywc_nmg_mexshare_payment_due }
				}
			}
			if = { limit = { NOT = { has_variable = ywc_nmg_mexshare_term_timer } NOT = { var:ywc_nmg_mexshare_ended_generation ?= var:ywc_nmg_mexshare_contract_generation } } ywc_nmg_mexshare_bump_serial = yes set_variable = { name = ywc_nmg_mexshare_ended_generation value = var:ywc_nmg_mexshare_contract_generation } }
		}
		if = { limit = { ywc_nmg_mexshare_agreement_current = yes ywc_nmg_v03_finance_safe = yes } add_modifier = { name = ywc_nmg_local_port_budget_right months = 1 } }
		else = { remove_modifier = ywc_nmg_local_port_budget_right }
	}
	if = { limit = { OR = { var:ywc_nmg_mexshare_state ?= 1 var:ywc_nmg_mexshare_state ?= 2 } OR = { NOT = { has_variable = ywc_nmg_mexshare_request_timer } NOT = { ywc_nmg_mexshare_link_intact = yes } } } ywc_nmg_mexshare_cancel_unsigned = yes }
}

ywc_nmg_mexshare_end_agreement = {
	if = {
		limit = { exists = scope:ywc_nmg_mexshare_end_owner scope:ywc_nmg_mexshare_end_owner ?= this has_variable = ywc_nmg_mexshare_contract_generation var:ywc_nmg_mexshare_contract_generation ?= scope:ywc_nmg_mexshare_end_generation NOT = { var:ywc_nmg_mexshare_ended_generation ?= scope:ywc_nmg_mexshare_end_generation } }
		ywc_nmg_mexshare_monthly = yes
		if = { limit = { NOT = { var:ywc_nmg_mexshare_ended_generation ?= scope:ywc_nmg_mexshare_end_generation } } ywc_nmg_mexshare_bump_serial = yes set_variable = { name = ywc_nmg_mexshare_ended_generation value = scope:ywc_nmg_mexshare_end_generation } }
		remove_variable = ywc_nmg_mexshare_term_timer
		remove_modifier = ywc_nmg_local_port_budget_right
	}
}
