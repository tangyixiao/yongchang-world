# NMG foundation scripted effects. All fees derive from real treasury capacity.

ywc_nmg_release_slot = {
	change_variable = { name = ywc_v03_active_arc_count add = -1 }
	change_variable = { name = ywc_v03_visible_task_count add = -1 }
	clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 }
	clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 }
}

ywc_nmg_land_fee_snapshot = {
	if = {
		limit = { NOT = { has_variable = ywc_nmg_land_fee } }
		set_variable = { name = ywc_nmg_land_fee value = yearly_gross_income }
		change_variable = { name = ywc_nmg_land_fee divide = 52 }
		clamp_variable = { name = ywc_nmg_land_fee min = 100 max = 1000000000 }
	}
}

ywc_nmg_land_pay_fee_once = {
	if = {
		limit = { NOT = { has_variable = ywc_nmg_land_fee } }
		ywc_nmg_land_fee_snapshot = yes
	}
	if = {
		limit = {
			NOT = { has_variable = ywc_nmg_land_fee_paid }
			gold_reserves >= var:ywc_nmg_land_fee
		}
		add_treasury = { value = var:ywc_nmg_land_fee multiply = -1 }
		set_variable = { name = ywc_nmg_land_fee_paid value = 1 }
	}
}

ywc_nmg_land_monthly_review = {
	if = {
		limit = {
			has_variable = ywc_v03_nmg_v03_nmg_land_arbitration_started
			NOT = { has_variable = ywc_v03_nmg_v03_nmg_land_arbitration_settled }
		}
		if = {
			limit = { NOT = { has_variable = ywc_nmg_land_watch_started } }
			set_variable = { name = ywc_nmg_land_watch_started value = 1 }
			set_variable = { name = ywc_nmg_land_watch_timer value = 1 months = 6 }
		}
		if = {
			limit = {
				has_variable = ywc_nmg_land_watch_started
				NOT = { has_variable = ywc_nmg_land_watch_timer }
				NOT = { has_variable = ywc_nmg_land_six_months_complete }
			}
			set_variable = { name = ywc_nmg_land_six_months_complete value = 1 }
			remove_variable = ywc_nmg_land_watch_started
		}
		if = {
			limit = {
				has_variable = ywc_nmg_land_six_months_complete
				NOT = { has_variable = ywc_nmg_land_12_notice }
			}
			set_variable = { name = ywc_nmg_land_12_notice value = 1 months = 12 }
		}
		if = {
			limit = {
				has_variable = ywc_nmg_land_six_months_complete
				NOT = { has_variable = ywc_nmg_land_36_notice }
			}
			set_variable = { name = ywc_nmg_land_36_notice value = 1 months = 36 }
		}
		if = {
			limit = {
				has_variable = ywc_nmg_land_six_months_complete
				NOT = { ywc_nmg_land_route_ready = yes }
			}
			ywc_nmg_land_note_blocked = yes
		}
	}
}

			limit = { NOT = { has_variable = ywc_nmg_revenue_watch_started } }
			set_variable = { name = ywc_nmg_revenue_watch_started value = 1 }
			set_variable = { name = ywc_nmg_revenue_watch_timer value = 1 months = 6 }
		}
		if = {
			limit = {
				has_variable = ywc_nmg_revenue_watch_started
				NOT = { has_variable = ywc_nmg_revenue_watch_timer }
				NOT = { has_variable = ywc_nmg_revenue_six_months_complete }
			}
			set_variable = { name = ywc_nmg_revenue_six_months_complete value = 1 }
			remove_variable = ywc_nmg_revenue_watch_started
		}
		if = {
			limit = {
				has_variable = ywc_nmg_revenue_six_months_complete
				NOT = { has_variable = ywc_nmg_revenue_12_notice }
			}
			set_variable = { name = ywc_nmg_revenue_12_notice value = 1 months = 12 }
		}
		if = {
			limit = {
				has_variable = ywc_nmg_revenue_six_months_complete
				NOT = { has_variable = ywc_nmg_revenue_36_notice }
			}
			set_variable = { name = ywc_nmg_revenue_36_notice value = 1 months = 36 }
		}
		if = {
			limit = {
				has_variable = ywc_nmg_revenue_six_months_complete
				NOT = { ywc_nmg_revenue_route_ready = yes }
			}
			ywc_nmg_revenue_note_blocked = yes
		}
	}
