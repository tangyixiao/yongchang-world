# Crisis C07 债务、矿权与公法: scripted effects. Every cross-country state
# transition writes BOTH sides inside one effect, each cross-scope write is
# guarded by an if-limit that re-checks the target's live state, and no
# effect removes ywc_c07_pending right before a gated trigger_event.

# Shared receipt: consume the pending marker, advance the popup epoch, save
# the epoch scope for the option at_ triggers, and start the 90-day
# major-event cooldown. Called only from event immediates whose limit
# already verified the delivery trigger, so the removal never precedes a
# gated trigger_event.
ywc_c07_receive_major = { remove_variable = ywc_c07_pending if = { limit = { NOT = { has_variable = ywc_c07_popup_epoch } } set_variable = { name = ywc_c07_popup_epoch value = 0 } } change_variable = { name = ywc_c07_popup_epoch add = 1 } save_scope_value_as = { name = ywc_c07_popup_epoch value = var:ywc_c07_popup_epoch } ywc_v03_mark_major_delivered = yes }

# Release this country's single crisis arc slot; only effective once.
ywc_c07_release_capacity = { if = { limit = { var:ywc_c07_started ?= 1 NOT = { has_variable = ywc_c07_released } } set_variable = { name = ywc_c07_released value = 1 } change_variable = { name = ywc_v03_active_arc_count add = -1 } change_variable = { name = ywc_v03_visible_task_count add = -1 } clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 } clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 } } }

# Segment markers used by the terminal closers only. They are written when
# the holder's flow is being wound up (pending consumed, no further
# delivery planned) so held journals can resolve instead of stalling.
ywc_c07_mark_segments_done = { set_variable = { name = ywc_c07_registration_done value = 1 } set_variable = { name = ywc_c07_execution_done value = 1 } }

# Terminal closers. Pending is cleared only here, where no further delivery
# depends on it, together with the final outcome and the slot release.
ywc_c07_close_status_quo = { set_variable = { name = ywc_c07_outcome value = 3 } set_variable = { name = ywc_c07_wait_age value = 0 } ywc_c07_mark_segments_done = yes if = { limit = { has_variable = ywc_c07_pending } remove_variable = ywc_c07_pending } ywc_c07_release_capacity = yes }
ywc_c07_close_withdrawn = { set_variable = { name = ywc_c07_outcome value = 4 } set_variable = { name = ywc_c07_wait_age value = 0 } ywc_c07_mark_segments_done = yes if = { limit = { has_variable = ywc_c07_pending } remove_variable = ywc_c07_pending } ywc_c07_release_capacity = yes }
