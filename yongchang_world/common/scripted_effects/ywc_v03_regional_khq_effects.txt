# Regional KHQ (F02) support effects. The league-compact journal opens only
# after the banner-land line has settled a real outcome (1 or 2), never at
# startup; a withdrawn banner-land line leaves it closed.
ywc_khq_open_league_je = {
	if = {
		limit = {
			NOT = { has_journal_entry = ywc_je_v03_khq_b }
			NOT = { has_variable = ywc_khq_b_started }
			is_country_alive = yes
			has_variable = ywc_khq_a_outcome
			NOT = { var:ywc_khq_a_outcome ?= 4 }
		}
		add_journal_entry = { type = ywc_je_v03_khq_b }
	}
}
