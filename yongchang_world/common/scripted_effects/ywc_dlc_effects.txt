# Every path starts with the no-DLC fallback, then conditionally adds one
# bounded enhancement. No path grants territory or permanent free military.

ywc_apply_dlc_compatibility = {
	# no DLC fallback: subjects, relations, variables and finite modifiers only
	set_variable = { name = ywc_dlc_base_path value = 1 }
	if = {
		limit = { has_dlc_feature = ep1_content }
		set_variable = { name = ywc_dlc_ep1_active value = 1 }
		add_modifier = { name = ywc_dlc_ep1_investment days = 3650 }
	}
	if = {
		limit = { has_dlc_feature = mp1_content }
		set_variable = { name = ywc_dlc_mp1_active value = 1 }
		add_modifier = { name = ywc_dlc_mp1_charters days = 3650 }
	}
	if = {
		limit = { has_dlc_feature = ep2_content }
		set_variable = { name = ywc_dlc_ep2_active value = 1 }
		add_modifier = { name = ywc_dlc_ep2_flagship days = 3650 }
	}
}
