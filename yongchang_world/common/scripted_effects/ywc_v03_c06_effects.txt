# Crisis C06 滇缅山口与双重贡赋: scripted effects. Every cross-country state
# transition writes BOTH sides inside one effect, each cross-scope write is
# guarded by an if-limit that re-checks the target's live state, and no effect
# removes ywc_c06_pending right before a gated trigger_event.

# Shared receipt: consume the pending marker, advance the popup epoch, save the
# epoch scope for the option at_ triggers, and start the 90-day major-event
# cooldown. Called only from event immediates whose limit already verified the
# delivery trigger, so the removal never precedes a gated trigger_event.
ywc_c06_receive_major = { remove_variable = ywc_c06_pending if = { limit = { NOT = { has_variable = ywc_c06_popup_epoch } } set_variable = { name = ywc_c06_popup_epoch value = 0 } } change_variable = { name = ywc_c06_popup_epoch add = 1 } save_scope_value_as = { name = ywc_c06_popup_epoch value = var:ywc_c06_popup_epoch } ywc_v03_mark_major_delivered = yes }

# Release this country's single crisis arc slot; only effective once.
ywc_c06_release_capacity = { if = { limit = { var:ywc_c06_started ?= 1 NOT = { has_variable = ywc_c06_released } } set_variable = { name = ywc_c06_released value = 1 } change_variable = { name = ywc_v03_active_arc_count add = -1 } change_variable = { name = ywc_v03_visible_task_count add = -1 } clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 } clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 } } }

# Terminal closers. Pending is cleared only here, where no further delivery
# depends on it, together with the final outcome and the slot release.
ywc_c06_close_status_quo = { set_variable = { name = ywc_c06_outcome value = 3 } set_variable = { name = ywc_c06_orders_done value = 1 } set_variable = { name = ywc_c06_wait_age value = 0 } if = { limit = { has_variable = ywc_c06_pending } remove_variable = ywc_c06_pending } ywc_c06_release_capacity = yes }
