# Crisis C04 南洋航路：港规、护航与不同国家的船: scripted effects. Every cross-country
# state transition writes BOTH sides inside one effect, each cross-scope write
# is guarded by an if-limit that re-checks the target's live state, and no
# effect removes ywc_c04_pending right before a gated trigger_event.

# Shared receipt: consume the pending marker, advance the popup epoch, save the
# epoch scope for the option at_ triggers, and start the 90-day major-event
# cooldown. Called only from event immediates whose limit already verified the
# delivery trigger, so the removal never precedes a gated trigger_event.
ywc_c04_receive_major = { remove_variable = ywc_c04_pending if = { limit = { NOT = { has_variable = ywc_c04_popup_epoch } } set_variable = { name = ywc_c04_popup_epoch value = 0 } } change_variable = { name = ywc_c04_popup_epoch add = 1 } save_scope_value_as = { name = ywc_c04_popup_epoch value = var:ywc_c04_popup_epoch } ywc_v03_mark_major_delivered = yes }

# Release this country's single crisis arc slot; only effective once.
ywc_c04_release_capacity = { if = { limit = { var:ywc_c04_started ?= 1 NOT = { has_variable = ywc_c04_released } } set_variable = { name = ywc_c04_released value = 1 } change_variable = { name = ywc_v03_active_arc_count add = -1 } change_variable = { name = ywc_v03_visible_task_count add = -1 } clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 } clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 } } }

# Organizer: mount the ore-ship seizure conflict. One island that has finished
# its inspection round (waiting stage 1321) seizes; the cargo-owner country
# (LAN mining exports, else DMG port trade, never the organizer itself)
# answers separately. The organizer remounts its own waiting stage in the
# same effect, or walks straight to the flag check when nobody can be mounted.
ywc_c04_mount_seizure = {
	if = {
		limit = { c:WBK ?= { is_country_alive = yes has_variable = ywc_c04_case_open NOT = { has_variable = ywc_c04_outcome } var:ywc_c04_stage ?= 1321 } }
		set_variable = { name = ywc_c04_seizure_island value = 1 }
		c:WBK = { set_variable = { name = ywc_c04_seizing value = 1 } set_variable = { name = ywc_c04_review_open value = 1 } set_variable = { name = ywc_c04_stage value = 1306 } set_variable = { name = ywc_c04_pending value = 1306 } set_variable = { name = ywc_c04_wait_age value = 0 } change_variable = { name = ywc_c04_popup_epoch add = 1 } }
	}
	else_if = {
		limit = { c:MHG ?= { is_country_alive = yes has_variable = ywc_c04_case_open NOT = { has_variable = ywc_c04_outcome } var:ywc_c04_stage ?= 1321 } }
		set_variable = { name = ywc_c04_seizure_island value = 2 }
		c:MHG = { set_variable = { name = ywc_c04_seizing value = 1 } set_variable = { name = ywc_c04_review_open value = 1 } set_variable = { name = ywc_c04_stage value = 1306 } set_variable = { name = ywc_c04_pending value = 1306 } set_variable = { name = ywc_c04_wait_age value = 0 } change_variable = { name = ywc_c04_popup_epoch add = 1 } }
	}
	if = {
		limit = { OR = { var:ywc_c04_role ?= 1 var:ywc_c04_role ?= 2 } c:LAN ?= { is_country_alive = yes has_variable = ywc_c04_case_open var:ywc_c04_escort_open ?= 1 NOT = { has_variable = ywc_c04_outcome } NOT = { var:ywc_c04_stage ?= 1314 } } }
		set_variable = { name = ywc_c04_seizure_owner value = 1 }
		c:LAN = { set_variable = { name = ywc_c04_review_open value = 1 } set_variable = { name = ywc_c04_stage value = 1314 } set_variable = { name = ywc_c04_pending value = 1314 } set_variable = { name = ywc_c04_wait_age value = 0 } change_variable = { name = ywc_c04_popup_epoch add = 1 } }
	}
	else_if = {
		limit = { OR = { var:ywc_c04_role ?= 1 var:ywc_c04_role ?= 3 } c:DMG ?= { is_country_alive = yes has_variable = ywc_c04_case_open var:ywc_c04_escort_open ?= 1 NOT = { has_variable = ywc_c04_outcome } NOT = { var:ywc_c04_stage ?= 1314 } } }
		set_variable = { name = ywc_c04_seizure_owner value = 2 }
		c:DMG = { set_variable = { name = ywc_c04_review_open value = 1 } set_variable = { name = ywc_c04_stage value = 1314 } set_variable = { name = ywc_c04_pending value = 1314 } set_variable = { name = ywc_c04_wait_age value = 0 } change_variable = { name = ywc_c04_popup_epoch add = 1 } }
	}
	if = { limit = { OR = { has_variable = ywc_c04_seizure_island has_variable = ywc_c04_seizure_owner } }
		set_variable = { name = ywc_c04_stage value = 1317 }
		set_variable = { name = ywc_c04_wait_age value = 0 }
		change_variable = { name = ywc_c04_popup_epoch add = 1 }
	}
	else = {
		set_variable = { name = ywc_c04_stage value = 1307 }
		set_variable = { name = ywc_c04_pending value = 1307 }
		set_variable = { name = ywc_c04_wait_age value = 0 }
		change_variable = { name = ywc_c04_popup_epoch add = 1 }
	}
}
