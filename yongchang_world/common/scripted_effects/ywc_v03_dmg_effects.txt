# Local DMG common-subjects helpers. All one-shot flags are checked before payment or counters change.
ywc_dmg_initialize_v03 = {
	if = { limit = { NOT = { has_variable = ywc_v03_initialized } } set_variable = { name = ywc_v03_initialized value = 1 } }
	if = { limit = { NOT = { has_variable = ywc_v03_active_arc_count } } set_variable = { name = ywc_v03_active_arc_count value = 0 } }
	if = { limit = { NOT = { has_variable = ywc_v03_visible_task_count } } set_variable = { name = ywc_v03_visible_task_count value = 0 } }
}

ywc_dmg_open_common_subjects_once = {
	ywc_dmg_initialize_v03 = yes
	if = {
		limit = {
			NOT = { has_variable = ywc_dmg_arc_open }
			var:ywc_v03_active_arc_count < 3
			var:ywc_v03_visible_task_count < 6
		}
		set_variable = { name = ywc_dmg_arc_open value = 1 }
		set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_started value = 1 }
		set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_last_event value = 0 }
		set_variable = { name = ywc_dmg_contract_stage value = 0 }
		set_variable = { name = ywc_dmg_review_due value = 1 months = 9 }
		set_variable = { name = ywc_dmg_blocked_12_due value = 1 months = 12 }
		set_variable = { name = ywc_dmg_blocked_36_due value = 1 months = 36 }
		change_variable = { name = ywc_v03_active_arc_count add = 1 }
		change_variable = { name = ywc_v03_visible_task_count add = 1 }
	}
}

ywc_dmg_snapshot_costs_once = {
	if = {
		limit = { NOT = { has_variable = ywc_dmg_costs_snapshotted } }
		set_variable = { name = ywc_dmg_public_registry_cost value = yearly_gross_income }
		change_variable = { name = ywc_dmg_public_registry_cost multiply = 0.0192307692308 }
		clamp_variable = { name = ywc_dmg_public_registry_cost min = 100 max = 1000000000 }
		set_variable = { name = ywc_dmg_compensation_cost value = var:ywc_dmg_public_registry_cost }
		set_variable = { name = ywc_dmg_status_orders_cost value = var:ywc_dmg_public_registry_cost }
		set_variable = { name = ywc_dmg_costs_snapshotted value = 1 }
	}
}

ywc_dmg_pay_public_registry_once = {
	ywc_dmg_snapshot_costs_once = yes
	if = {
		limit = { NOT = { has_variable = ywc_dmg_public_registry_paid } }
		add_treasury = { value = var:ywc_dmg_public_registry_cost multiply = -1 }
		set_variable = { name = ywc_dmg_public_registry_paid value = 1 }
	}
}

	remove_modifier = ywc_dmg_language_central_forms
	remove_modifier = ywc_dmg_language_translation
	remove_modifier = ywc_dmg_language_old_register
	remove_modifier = ywc_dmg_precedent_central
	remove_modifier = ywc_dmg_precedent_local_courts

		limit = { var:ywc_dmg_public_language_mode = 1 }
		add_modifier = { name = ywc_dmg_language_central_forms months = 24 }
	}
	else_if = {
		limit = { var:ywc_dmg_public_language_mode = 2 }

	}
	else = { add_modifier = { name = ywc_dmg_language_old_register months = 24 } }

	}
	else = { add_modifier = { name = ywc_dmg_precedent_lineage months = 24 } }

		set_variable = { name = ywc_dmg_appointment_actual value = 1 }
	}
}

ywc_dmg_close_as_common = {
	if = {
		limit = { NOT = { has_variable = ywc_dmg_any_outcome } }
		ywc_dmg_record_appointment_actual = yes
		if = {
			limit = { has_variable = ywc_dmg_appointment_actual bureaucracy >= 0 }
			set_variable = { name = ywc_dmg_any_outcome value = 1 }
			set_variable = { name = ywc_dmg_outcome_common_qualification value = 1 }
			set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_outcome value = 1 }
		}
		else = {
			set_variable = { name = ywc_dmg_any_outcome value = 1 }
			set_variable = { name = ywc_dmg_outcome_limited_qualification value = 1 }
			set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_outcome value = 5 }
		}
	}
	ywc_v03_release_arc_dmg_v03_dmg_common_subjects = yes
}

ywc_v03_bridge_dmg_compact_one = {
	if = {
		limit = { NOT = { has_variable = ywc_campaign_dmg_c1_outcome } }
		if = {
			limit = { var:ywc_v03_dmg_v03_dmg_common_subjects_outcome = 1 }
			set_variable = { name = ywc_campaign_dmg_c1_outcome value = 1 }
			set_variable = { name = ywc_je_flavor_dmg_compact_1_resolved value = 1 }
		}
		else_if = {
			limit = { OR = { var:ywc_v03_dmg_v03_dmg_common_subjects_outcome = 2 var:ywc_v03_dmg_v03_dmg_common_subjects_outcome = 3 } }
			set_variable = { name = ywc_campaign_dmg_c1_outcome value = 2 }
			set_variable = { name = ywc_je_flavor_dmg_compact_1_resolved value = 1 }
		}
		else = {
			set_variable = { name = ywc_campaign_dmg_c1_outcome value = 3 }
			set_variable = { name = ywc_je_flavor_dmg_compact_1_failed value = 1 }
		}
	}
	if = {
		limit = {
			NOT = { has_journal_entry = ywc_je_flavor_dmg_compact_2 }
			NOT = { has_variable = ywc_je_flavor_dmg_compact_2_resolved }
			NOT = { has_variable = ywc_je_flavor_dmg_compact_2_failed }
		}
		add_journal_entry = { type = ywc_je_flavor_dmg_compact_2 }
	}
}

ywc_v03_release_arc_dmg_v03_dmg_common_subjects = {
	if = {
		limit = {
			has_variable = ywc_dmg_arc_open
			NOT = { has_variable = ywc_v03_dmg_v03_dmg_common_subjects_settled }
			has_variable = ywc_dmg_any_outcome
		}
		set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_settled value = 1 }
		change_variable = { name = ywc_v03_active_arc_count add = -1 }
		change_variable = { name = ywc_v03_visible_task_count add = -1 }
		clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 }
		clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 }
		ywc_v03_bridge_dmg_compact_one = yes
	}
}
