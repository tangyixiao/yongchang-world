# Local DMG common-subjects helpers. All one-shot flags are checked before payment or counters change.
ywc_dmg_initialize_v03 = {
	if = { limit = { NOT = { has_variable = ywc_v03_initialized } } set_variable = { name = ywc_v03_initialized value = 1 } }
	if = { limit = { NOT = { has_variable = ywc_v03_active_arc_count } } set_variable = { name = ywc_v03_active_arc_count value = 0 } }
	if = { limit = { NOT = { has_variable = ywc_v03_visible_task_count } } set_variable = { name = ywc_v03_visible_task_count value = 0 } }
}

ywc_dmg_open_common_subjects_once = {
	ywc_dmg_initialize_v03 = yes
	ywc_dmg_snapshot_costs_once = yes
	if = {
		limit = {
			NOT = { has_variable = ywc_dmg_arc_open }
			var:ywc_v03_active_arc_count < 3
			var:ywc_v03_visible_task_count < 6
		}
		set_variable = { name = ywc_dmg_arc_open value = 1 }
		set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_started value = 1 }
		set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_last_event value = 0 }
		set_variable = { name = ywc_dmg_contract_stage value = 0 }
		set_variable = { name = ywc_dmg_blocked_12_due value = 1 months = 12 }
		set_variable = { name = ywc_dmg_blocked_36_due value = 1 months = 36 }
		change_variable = { name = ywc_v03_active_arc_count add = 1 }
		change_variable = { name = ywc_v03_visible_task_count add = 1 }
	}
}

ywc_dmg_snapshot_costs_once = {
	if = {
		limit = { NOT = { has_variable = ywc_dmg_costs_snapshotted } }
		set_variable = { name = ywc_dmg_public_registry_cost value = yearly_gross_income }
		change_variable = { name = ywc_dmg_public_registry_cost divide = 52 }
		clamp_variable = { name = ywc_dmg_public_registry_cost min = 100 max = 1000000000 }
		set_variable = { name = ywc_dmg_compensation_cost value = var:ywc_dmg_public_registry_cost }
		set_variable = { name = ywc_dmg_status_orders_cost value = var:ywc_dmg_public_registry_cost }
		set_variable = { name = ywc_dmg_costs_snapshotted value = 1 }
	}
}

# Start remedy clocks only while the selected route is genuinely blocked. A
# restored route clears stale notices and allows a later interruption to start
# a fresh continuous interval.
ywc_dmg_foundation_update_titles_blocked_clock = {
	if = { limit = { ywc_dmg_foundation_titles_route_ready = yes }
		remove_variable = ywc_dmg_titles_blocked_clock_started
		remove_variable = ywc_dmg_titles_blocked_12_due
		remove_variable = ywc_dmg_titles_blocked_36_due
		remove_variable = ywc_dmg_titles_blocked_12_notice
		remove_variable = ywc_dmg_titles_blocked_36_notice
		remove_variable = ywc_dmg_titles_blocked_12_delivered
		if = { limit = { NOT = { AND = { has_variable = ywc_dmg_titles_lower_target_proposed has_variable = ywc_dmg_titles_lower_target_selected var:ywc_dmg_titles_route = 3 has_variable = ywc_dmg_titles_newcomer_mode var:ywc_dmg_titles_newcomer_mode = 3 } } } remove_variable = ywc_dmg_titles_blocked_36_delivered }
		remove_variable = ywc_dmg_titles_emergency_notice
	}
	else = {
		if = { limit = { NOT = { has_variable = ywc_dmg_titles_blocked_clock_started } NOT = { has_variable = ywc_dmg_titles_outcome } }
			set_variable = { name = ywc_dmg_titles_blocked_clock_started value = 1 }
			set_variable = { name = ywc_dmg_titles_blocked_12_due value = 1 months = 12 }
			set_variable = { name = ywc_dmg_titles_blocked_36_due value = 1 months = 36 }
		}
		if = { limit = { has_variable = ywc_dmg_titles_blocked_clock_started NOT = { has_variable = ywc_dmg_titles_outcome } NOT = { has_variable = ywc_dmg_titles_blocked_12_due } NOT = { has_variable = ywc_dmg_titles_blocked_12_delivered } }
			set_variable = { name = ywc_dmg_titles_blocked_12_notice value = 1 }
		}
		if = { limit = { has_variable = ywc_dmg_titles_blocked_clock_started NOT = { has_variable = ywc_dmg_titles_outcome } NOT = { has_variable = ywc_dmg_titles_blocked_36_due } NOT = { has_variable = ywc_dmg_titles_blocked_36_delivered } }
			set_variable = { name = ywc_dmg_titles_blocked_36_notice value = 1 }
		}
	}
	if = { limit = { NOT = { ywc_dmg_v03_has_owned_state = yes } NOT = { has_variable = ywc_dmg_titles_outcome } }
		set_variable = { name = ywc_dmg_titles_emergency_notice value = 1 }
	}
	else = { remove_variable = ywc_dmg_titles_emergency_notice }
}
