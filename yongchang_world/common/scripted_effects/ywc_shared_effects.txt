ywc_set_heritage_legitimacy = {
    set_variable = { name = ywc_heritage_legitimacy value = 50 }
    # clamp = { min = 0 max = 100 }
    clamp_variable = { name = ywc_heritage_legitimacy min = 0 max = 100 }
    ywc_refresh_heritage_legitimacy_modifier = yes
}

ywc_add_heritage_legitimacy = {
    change_variable = { name = ywc_heritage_legitimacy add = 5 }
    clamp_variable = { name = ywc_heritage_legitimacy min = 0 max = 100 }
    ywc_refresh_heritage_legitimacy_modifier = yes
}

ywc_lower_heritage_legitimacy = {
    change_variable = { name = ywc_heritage_legitimacy add = -10 }
    clamp_variable = { name = ywc_heritage_legitimacy min = 0 max = 100 }
    ywc_refresh_heritage_legitimacy_modifier = yes
}

ywc_refresh_heritage_legitimacy_modifier = {
    remove_modifier = ywc_heritage_legitimacy_high
    remove_modifier = ywc_heritage_legitimacy_low
    if = {
        limit = { var:ywc_heritage_legitimacy >= 70 }
        add_modifier = { name = ywc_heritage_legitimacy_high }
    }
    else_if = {
        limit = { var:ywc_heritage_legitimacy <= 30 }
        add_modifier = { name = ywc_heritage_legitimacy_low }
    }
}
