# Shared v0.3 lifecycle counters. Country-specific JE conditions remain in
# each catalog-generated JE; these effects only own version and active slots.

ywc_v03_initialize_country = {
	if = {
		limit = { NOT = { has_variable = ywc_v03_initialized } }
		set_variable = { name = ywc_v03_initialized value = 1 }
	}
	if = {
		limit = { NOT = { has_variable = ywc_v03_release_version } }
		set_variable = { name = ywc_v03_release_version value = 3 }
	}
	if = {
		limit = { NOT = { has_variable = ywc_v03_active_arc_count } }
		set_variable = { name = ywc_v03_active_arc_count value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = ywc_v03_visible_task_count } }
		set_variable = { name = ywc_v03_visible_task_count value = 0 }
	}
	ywc_v03_register_legacy_continuation = yes
}

# Per-arc compatibility: 1 = new entry allowed, 2 = old chapter continues,
# 3 = old chapter terminal. Skipping a duplicate opening is not reform success.
# A new arc already started or settled always retains its own state.
ywc_v03_register_legacy_continuation = {
	if = {
		limit = {
			c:SHU ?= this
			NOT = { has_variable = ywc_v03_shu_v03_shu_fiscal_register_started }
			NOT = { has_variable = ywc_v03_shu_v03_shu_fiscal_register_settled }
		}
		if = {
			limit = {
				OR = {
					has_variable = ywc_campaign_shu_c1_outcome
					has_variable = ywc_je_flavor_shu_court_1_resolved
					has_variable = ywc_je_flavor_shu_court_1_failed
				}
			}
			set_variable = { name = ywc_v03_shu_v03_shu_fiscal_register_compatibility value = 3 }
			set_variable = { name = ywc_v03_shu_v03_shu_fiscal_register_settled value = 1 }
		}
		else_if = {
			limit = {
				OR = {
					has_variable = ywc_campaign_shu_c1_prelude
					has_variable = ywc_campaign_shu_c1_score
					has_variable = ywc_campaign_shu_c1_e100_done
					has_variable = ywc_campaign_shu_c1_e101_done
					has_variable = ywc_campaign_shu_c1_e102_done
					has_variable = ywc_campaign_shu_c1_e103_done
					has_variable = ywc_campaign_shu_c1_e104_done
				}
			}
			set_variable = { name = ywc_v03_shu_v03_shu_fiscal_register_compatibility value = 2 }
			set_variable = { name = ywc_v03_shu_v03_shu_fiscal_register_settled value = 1 }
		}
		else = { set_variable = { name = ywc_v03_shu_v03_shu_fiscal_register_compatibility value = 1 } }
	}
	if = {
		limit = {
			c:DMG ?= this
			NOT = { has_variable = ywc_v03_dmg_v03_dmg_common_subjects_started }
			NOT = { has_variable = ywc_v03_dmg_v03_dmg_common_subjects_settled }
		}
		if = {
			limit = {
				OR = {
					has_variable = ywc_campaign_dmg_c1_outcome
					has_variable = ywc_je_flavor_dmg_compact_1_resolved
					has_variable = ywc_je_flavor_dmg_compact_1_failed
				}
			}
			set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_compatibility value = 3 }
			set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_settled value = 1 }
		}
		else_if = {
			limit = {
				OR = {
					has_variable = ywc_campaign_dmg_c1_prelude
					has_variable = ywc_campaign_dmg_c1_score
					has_variable = ywc_campaign_dmg_c1_e100_done
					has_variable = ywc_campaign_dmg_c1_e101_done
					has_variable = ywc_campaign_dmg_c1_e102_done
					has_variable = ywc_campaign_dmg_c1_e103_done
					has_variable = ywc_campaign_dmg_c1_e104_done
				}
			}
			set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_compatibility value = 2 }
			set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_settled value = 1 }
		}
		else = { set_variable = { name = ywc_v03_dmg_v03_dmg_common_subjects_compatibility value = 1 } }
	}
	if = {
		limit = {
			c:KOR ?= this
			NOT = { has_variable = ywc_v03_kor_v03_kor_military_accounts_started }
			NOT = { has_variable = ywc_v03_kor_v03_kor_military_accounts_settled }
		}
		if = {
			limit = {
				OR = {
					has_variable = ywc_campaign_kor_c1_outcome
					has_variable = ywc_je_flavor_kor_reform_1_resolved
					has_variable = ywc_je_flavor_kor_reform_1_failed
				}
			}
			set_variable = { name = ywc_v03_kor_v03_kor_military_accounts_compatibility value = 3 }
			set_variable = { name = ywc_v03_kor_v03_kor_military_accounts_settled value = 1 }
		}
		else_if = {
			limit = {
				OR = {
					has_variable = ywc_campaign_kor_c1_prelude
					has_variable = ywc_campaign_kor_c1_score
					has_variable = ywc_campaign_kor_c1_e100_done
					has_variable = ywc_campaign_kor_c1_e101_done
					has_variable = ywc_campaign_kor_c1_e102_done
					has_variable = ywc_campaign_kor_c1_e103_done
					has_variable = ywc_campaign_kor_c1_e104_done
				}
			}
			set_variable = { name = ywc_v03_kor_v03_kor_military_accounts_compatibility value = 2 }
			set_variable = { name = ywc_v03_kor_v03_kor_military_accounts_settled value = 1 }
		}
		else = { set_variable = { name = ywc_v03_kor_v03_kor_military_accounts_compatibility value = 1 } }
	}
}

# Invoke only after a main-arc event is actually delivered; duplicate delivery
# handlers must not extend an already-running 90-day quiet period.
ywc_v03_mark_major_delivered = {
	if = {
		limit = { NOT = { has_variable = ywc_v03_major_event_cooldown } }
		set_variable = { name = ywc_v03_major_event_cooldown value = 1 days = 90 }
	}
}

# Called by the first JE's immediate effect, after its possible trigger has
# verified capacity. The second guard makes the claim safe under manual entry.
ywc_v03_claim_arc_shu_v03_shu_fiscal_register = {
	ywc_v03_initialize_country = yes
	if = {
		limit = {
			NOT = { has_variable = ywc_v03_shu_v03_shu_fiscal_register_started }
			var:ywc_v03_active_arc_count < 3
			var:ywc_v03_visible_task_count < 6
		}
		set_variable = { name = ywc_v03_shu_v03_shu_fiscal_register_started value = 1 }
		set_variable = { name = ywc_v03_shu_v03_shu_fiscal_register_last_event value = 0 }
		change_variable = { name = ywc_v03_active_arc_count add = 1 }
		change_variable = { name = ywc_v03_visible_task_count add = 1 }
	}
}

# Called only after the country-specific result has been computed and stored.
# The bridge maps that actual result into the legacy chapter-one interface.
ywc_v03_bridge_shu_court_one = {
	if = {
		limit = { NOT = { has_variable = ywc_campaign_shu_c1_outcome } }
		if = {
			limit = { var:ywc_v03_shu_v03_shu_fiscal_register_outcome = 1 }
			set_variable = { name = ywc_campaign_shu_c1_outcome value = 1 }
			set_variable = { name = ywc_je_flavor_shu_court_1_resolved value = 1 }
		}
		else_if = {
			limit = { var:ywc_v03_shu_v03_shu_fiscal_register_outcome <= 3 }
			set_variable = { name = ywc_campaign_shu_c1_outcome value = 2 }
			set_variable = { name = ywc_je_flavor_shu_court_1_resolved value = 1 }
		}
		else = {
			set_variable = { name = ywc_campaign_shu_c1_outcome value = 3 }
			set_variable = { name = ywc_je_flavor_shu_court_1_failed value = 1 }
		}
	}
	if = {
		limit = {
			NOT = { has_journal_entry = ywc_je_flavor_shu_court_2 }
			NOT = { has_variable = ywc_je_flavor_shu_court_2_resolved }
			NOT = { has_variable = ywc_je_flavor_shu_court_2_failed }
		}
		add_journal_entry = { type = ywc_je_flavor_shu_court_2 }
	}
}

ywc_v03_release_arc_shu_v03_shu_fiscal_register = {
	if = {
		limit = {
			has_variable = ywc_v03_shu_v03_shu_fiscal_register_started
			NOT = { has_variable = ywc_v03_shu_v03_shu_fiscal_register_settled }
			var:ywc_v03_shu_v03_shu_fiscal_register_outcome > 0
		}
		set_variable = { name = ywc_v03_shu_v03_shu_fiscal_register_settled value = 1 }
		change_variable = { name = ywc_v03_active_arc_count add = -1 }
		change_variable = { name = ywc_v03_visible_task_count add = -1 }
		clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 }
		clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 }
		ywc_v03_bridge_shu_court_one = yes
	}
}

# M3 SHU provincial contract: source-owned helpers.
ywc_v03_claim_arc_shu_v03_shu_provincial_contract = {
	ywc_v03_initialize_country = yes
	if = {
		limit = {
			NOT = { has_variable = ywc_v03_shu_v03_shu_provincial_contract_started }
			NOT = { has_variable = ywc_v03_shu_v03_shu_provincial_contract_settled }
			var:ywc_v03_active_arc_count < 3
			var:ywc_v03_visible_task_count < 6
		}
		set_variable = { name = ywc_v03_shu_v03_shu_provincial_contract_started value = 1 }
		change_variable = { name = ywc_v03_active_arc_count add = 1 }
		change_variable = { name = ywc_v03_visible_task_count add = 1 }
	}
}

ywc_v03_release_arc_shu_v03_shu_provincial_contract = {
	if = {
		limit = {
			has_variable = ywc_v03_shu_v03_shu_provincial_contract_started
			NOT = { has_variable = ywc_v03_shu_v03_shu_provincial_contract_settled }
			has_variable = ywc_shu_prov_outcome
			var:ywc_shu_prov_outcome > 0
		}
		set_variable = { name = ywc_v03_shu_v03_shu_provincial_contract_settled value = 1 }
		set_variable = { name = ywc_v03_shu_v03_shu_provincial_contract_outcome value = var:ywc_shu_prov_outcome }
		remove_variable = ywc_shu_prov_pending_event
remove_variable = ywc_shu_prov_review_running
		change_variable = { name = ywc_v03_active_arc_count add = -1 }
		change_variable = { name = ywc_v03_visible_task_count add = -1 }
		clamp_variable = { name = ywc_v03_active_arc_count min = 0 max = 3 }
		clamp_variable = { name = ywc_v03_visible_task_count min = 0 max = 6 }
	}
}
