ywc_lan_mine_monthly_review = {
 if = { limit = { NOT = { has_variable = ywc_lan_mine_established } NOT = { has_variable = ywc_lan_mine_compromise } NOT = { has_variable = ywc_lan_mine_withdrawn } }
  if = { limit = { ywc_lan_mine_target_ready = yes }
   remove_variable = ywc_lan_mine_blocked_months
   remove_variable = ywc_lan_mine_notice_event
   remove_variable = ywc_lan_mine_notice12_sent
   remove_variable = ywc_lan_mine_notice36_sent
   remove_variable = ywc_lan_mine_notice_current
   if = { limit = { has_variable = ywc_lan_mine_implementation_complete }
    if = { limit = { has_variable = ywc_lan_mine_observation_started NOT = { has_variable = ywc_lan_mine_observation_timer } } set_variable = { name = ywc_lan_mine_six_months_complete value = 1 } remove_variable = ywc_lan_mine_observation_started }
    if = { limit = { NOT = { has_variable = ywc_lan_mine_observation_started } NOT = { has_variable = ywc_lan_mine_six_months_complete } } set_variable = { name = ywc_lan_mine_observation_timer value = 1 months = 6 } set_variable = { name = ywc_lan_mine_observation_started value = 1 } }
   }
  }
  else = {
   remove_variable = ywc_lan_mine_observation_timer
   remove_variable = ywc_lan_mine_observation_started
   remove_variable = ywc_lan_mine_six_months_complete
   if = { limit = { has_variable = ywc_lan_mine_current_deferred ywc_lan_mine_current_obligation_ready = yes ywc_lan_mine_can_pay = yes }
    remove_variable = ywc_lan_mine_current_deferred
    remove_variable = ywc_lan_mine_blocked_months
    remove_variable = ywc_lan_mine_notice_event
    remove_variable = ywc_lan_mine_notice12_sent
    remove_variable = ywc_lan_mine_notice36_sent
    remove_variable = ywc_lan_mine_notice_current
   }
   else_if = { limit = { OR = { has_variable = ywc_lan_mine_implementation_complete AND = { has_variable = ywc_lan_mine_filing_fee_paid NOT = { ywc_lan_mine_can_pay = yes } } ywc_lan_mine_current_obligation_blocked = yes } }
    change_variable = { name = ywc_lan_mine_blocked_months add = 1 }
    if = { limit = { var:ywc_lan_mine_blocked_months >= 12 NOT = { has_variable = ywc_lan_mine_notice12_sent } } set_variable = { name = ywc_lan_mine_notice_event value = 12 } }
    if = { limit = { var:ywc_lan_mine_blocked_months >= 36 NOT = { has_variable = ywc_lan_mine_notice36_sent } } set_variable = { name = ywc_lan_mine_notice_event value = 36 } }
   }
   else = { remove_variable = ywc_lan_mine_blocked_months remove_variable = ywc_lan_mine_notice_event remove_variable = ywc_lan_mine_notice12_sent remove_variable = ywc_lan_mine_notice36_sent remove_variable = ywc_lan_mine_notice_current }
  }
 }
}

# can never silently substitute. No state object is created.
ywc_lan_gold_capture_pilot = {
	if = {
		limit = { ywc_lan_gold_open = yes NOT = { has_variable = ywc_lan_gold_pilot_state } }
		every_scope_state = {
			limit = { owner = ROOT state_region = s:STATE_WEST_BORNEO }
			save_scope_as = ywc_lan_gold_pilot_scope
		}
		if = { limit = { exists = scope:ywc_lan_gold_pilot_scope } set_variable = { name = ywc_lan_gold_pilot_state value = scope:ywc_lan_gold_pilot_scope } }
	}
}

# Seed payment: the project's real preparation envelope. One opening weekly
# gross income, clamped, captured once; a genuine mine-audit precedent or a
# finished public mandate cuts this specific first payment in half.
ywc_lan_gold_capture_seed = {
	if = {
		limit = { NOT = { has_variable = ywc_lan_gold_seed_unit } }
		set_variable = { name = ywc_lan_gold_seed_unit value = yearly_gross_income }
		change_variable = { name = ywc_lan_gold_seed_unit divide = 52 }
		clamp_variable = { name = ywc_lan_gold_seed_unit min = 100 max = 1000000000 }
	}
}

ywc_lan_gold_seed_due = {
	ywc_lan_gold_capture_seed = yes
	set_variable = { name = ywc_lan_gold_seed_due_value value = var:ywc_lan_gold_seed_unit }
	if = { limit = { ywc_lan_gold_seed_rate_half = yes } change_variable = { name = ywc_lan_gold_seed_due_value multiply = 0.5 } }
	clamp_variable = { name = ywc_lan_gold_seed_due_value min = 50 max = 1000000000 }
}

ywc_lan_gold_pay_seed = {
	if = { limit = { ywc_lan_gold_open = yes NOT = { has_variable = ywc_lan_gold_seed_paid } ywc_lan_gold_can_pay = yes ywc_lan_gold_pilot_owned = yes ywc_lan_gold_at_1120 = yes ywc_lan_gold_seed_affordable = yes }
		ywc_lan_gold_seed_due = yes
		if = { limit = { gold_reserves >= var:ywc_lan_gold_seed_due_value in_default = no NOT = { has_modifier = declared_bankruptcy } }
			add_treasury = { value = var:ywc_lan_gold_seed_due_value multiply = -1 }
			set_variable = { name = ywc_lan_gold_seed_paid value = 1 }
			set_variable = { name = ywc_lan_gold_seed_paid_amount value = var:ywc_lan_gold_seed_due_value }
			if = { limit = { ywc_lan_gold_seed_rate_half = yes } set_variable = { name = ywc_lan_gold_seed_source value = 1 } }
			set_variable = { name = ywc_lan_gold_stage value = 1121 }
			set_variable = { name = ywc_lan_gold_pending_event value = 1121 }
			set_variable = { name = ywc_v03_lan_v03_lan_beyond_gold_last_event value = 0 }
		}
	}
}

ywc_lan_gold_choose_route = {

# Basis payment: the real second instalment, priced from the same opening
# envelope. Two genuinely earned prior decisions change this later node:
#   * a monopoly grounded in the established public mine audit (source_precedent)
#     draws on audited mine revenue and pays a quarter;
#   * a finished public mandate (source_leverage) already holds the processing
#     books and pays three quarters for processing.
# Absent either fact, the honest full/half rates apply unchanged.
ywc_lan_gold_basis_rate_calc = {
	if = { limit = { ywc_lan_gold_source_leverage = yes var:ywc_lan_gold_route = 1 } set_variable = { name = ywc_lan_gold_basis_rate value = 1 } }
	else_if = { limit = { ywc_lan_gold_source_precedent = yes var:ywc_lan_gold_route = 3 } set_variable = { name = ywc_lan_gold_basis_rate value = 4 } }
	else_if = { limit = { OR = { var:ywc_lan_gold_route = 2 var:ywc_lan_gold_route = 3 } } set_variable = { name = ywc_lan_gold_basis_rate value = 2 } }
	else = { set_variable = { name = ywc_lan_gold_basis_rate value = 3 } }
}

ywc_lan_gold_basis_due = {
	ywc_lan_gold_capture_seed = yes
	if = {
		limit = { NOT = { has_variable = ywc_lan_gold_basis_due_value } }
		ywc_lan_gold_basis_rate_calc = yes
		set_variable = { name = ywc_lan_gold_basis_due_value value = var:ywc_lan_gold_seed_unit }
		if = { limit = { var:ywc_lan_gold_basis_rate = 1 } change_variable = { name = ywc_lan_gold_basis_due_value multiply = 0.75 } }
		else_if = { limit = { var:ywc_lan_gold_basis_rate = 2 } change_variable = { name = ywc_lan_gold_basis_due_value multiply = 0.5 } }
		else_if = { limit = { var:ywc_lan_gold_basis_rate = 4 } change_variable = { name = ywc_lan_gold_basis_due_value multiply = 0.25 } }
		clamp_variable = { name = ywc_lan_gold_basis_due_value min = 50 max = 1000000000 }
	}
}

ywc_lan_gold_pay_basis = {
	if = { limit = { ywc_lan_gold_open = yes has_variable = ywc_lan_gold_seed_paid var:ywc_lan_gold_seed_paid = 1 NOT = { has_variable = ywc_lan_gold_basis_paid } ywc_lan_gold_can_pay = yes ywc_lan_gold_route_chosen = yes ywc_lan_gold_pilot_owned = yes ywc_lan_gold_at_1122 = yes }
		ywc_lan_gold_basis_due = yes
		if = { limit = { gold_reserves >= var:ywc_lan_gold_basis_due_value in_default = no NOT = { has_modifier = declared_bankruptcy } }
			add_treasury = { value = var:ywc_lan_gold_basis_due_value multiply = -1 }
			set_variable = { name = ywc_lan_gold_basis_paid value = 1 }
			set_variable = { name = ywc_lan_gold_basis_paid_amount value = var:ywc_lan_gold_basis_due_value }
			if = { limit = { var:ywc_lan_gold_basis_rate = 1 } set_variable = { name = ywc_lan_gold_basis_source_leverage value = 1 } }
			else_if = { limit = { var:ywc_lan_gold_basis_rate = 4 } set_variable = { name = ywc_lan_gold_basis_source_precedent value = 1 } }
			set_variable = { name = ywc_lan_gold_stage value = 1123 }
			set_variable = { name = ywc_lan_gold_pending_event value = 1123 }
			set_variable = { name = ywc_v03_lan_v03_lan_beyond_gold_last_event value = 0 }
		}
	}
}

# Responsibility is named on the captured pilot. A staffed office gives a full
# arrangement; without staff only the current gap can be recorded (bounded).
# A finished public mandate (source_leverage) already leaves a named, staffed
# public budget office, so it can carry a full arrangement even when the local
# bureaucrat count is short -- a second, distinct later benefit of that fact.
ywc_lan_gold_record_responsible = {
	if = { limit = { ywc_lan_gold_open = yes has_variable = ywc_lan_gold_basis_paid var:ywc_lan_gold_basis_paid = 1 NOT = { has_variable = ywc_lan_gold_responsible_recorded } ywc_lan_gold_pilot_owned = yes ywc_lan_gold_at_1123 = yes }
		if = { limit = { ywc_lan_gold_responsible_full = yes } set_variable = { name = ywc_lan_gold_responsible_recorded value = 2 } set_variable = { name = ywc_lan_gold_responsible_tier_full value = 1 } }
		else_if = { limit = { ywc_lan_gold_office = yes } set_variable = { name = ywc_lan_gold_responsible_recorded value = 1 } }
		set_variable = { name = ywc_lan_gold_stage value = 1124 }
		set_variable = { name = ywc_lan_gold_pending_event value = 1124 }
		set_variable = { name = ywc_v03_lan_v03_lan_beyond_gold_last_event value = 0 }
	}
}

# Starts the continuous audit with an explicit one-day gate so the creation
# month is never counted as a completed observation. A mature observation is
# never restarted.
ywc_lan_gold_start_review = {
	if = {
		limit = { ywc_lan_gold_open = yes has_variable = ywc_lan_gold_basis_paid var:ywc_lan_gold_basis_paid = 1 has_variable = ywc_lan_gold_responsible_recorded NOT = { has_variable = ywc_lan_gold_observation_done } ywc_lan_gold_at_1124 = yes }
		remove_variable = ywc_lan_gold_observation_interrupted
		remove_variable = ywc_lan_gold_observation_done
		remove_variable = ywc_lan_gold_observation_waiting
		set_variable = { name = ywc_lan_gold_review_streak value = 0 }
		set_variable = { name = ywc_lan_gold_review_started value = 1 }
		set_variable = { name = ywc_lan_gold_observation_running value = 1 }
		set_variable = { name = ywc_lan_gold_observation_ticked value = 1 days = 1 }
		set_variable = { name = ywc_lan_gold_stage value = 1125 }
		set_variable = { name = ywc_v03_lan_v03_lan_beyond_gold_last_event value = 0 }
		remove_variable = ywc_lan_gold_pending_event
	}
}

# Called once per real monthly pulse. The one-day gate consumes itself so that
# duplicate same-day callbacks cannot count twice. A failing month zeroes the
# streak, pauses the loop and records an interruption that must be explicitly
# restarted. Only six consecutive valid months mature the audit.
ywc_lan_gold_track_review = {
	if = {
		limit = { ywc_lan_gold_open = yes var:ywc_lan_gold_stage = 1125 has_variable = ywc_lan_gold_observation_running NOT = { has_variable = ywc_lan_gold_observation_ticked } }
		set_variable = { name = ywc_lan_gold_observation_ticked value = 1 days = 1 }
		if = { limit = { ywc_lan_gold_pilot_lost = yes }
			set_variable = { name = ywc_lan_gold_review_streak value = 0 }
			remove_variable = ywc_lan_gold_observation_running
			remove_variable = ywc_lan_gold_observation_done
			set_variable = { name = ywc_lan_gold_observation_interrupted value = 1 }
		}
		else_if = { limit = { NOT = { ywc_lan_gold_current_conditions = yes } }
			set_variable = { name = ywc_lan_gold_review_streak value = 0 }
			remove_variable = ywc_lan_gold_observation_running
			remove_variable = ywc_lan_gold_observation_done
			set_variable = { name = ywc_lan_gold_observation_interrupted value = 1 }
		}
		else_if = { limit = { NOT = { has_variable = ywc_lan_gold_observation_done } }
			change_variable = { name = ywc_lan_gold_review_streak add = 1 }
			if = { limit = { var:ywc_lan_gold_review_streak >= 6 } set_variable = { name = ywc_lan_gold_observation_done value = 1 } remove_variable = ywc_lan_gold_observation_running }
		}
	}
}

# If the originally captured pilot state is lost, do NOT settle silently. Record
# the real loss, keep the journal open and queue the honest setback half of 1126
# so the player actually receives and chooses it. No reward, restoration, new
# pilot or dereference of the vanished state is attempted. Costs are untouched.
ywc_lan_gold_handle_pilot_loss = {
	if = {
		limit = { ywc_lan_gold_loss_pending = yes NOT = { has_variable = ywc_lan_gold_loss_observed } }
		set_variable = { name = ywc_lan_gold_loss_observed value = 1 }
		set_variable = { name = ywc_lan_gold_loss_stage value = 1126 }
		set_variable = { name = ywc_lan_gold_pending_event value = 1126 }
	}
}

# Ends. Established / compromise only when the current facts already satisfy
# the matching predicate; setback when they do not. Each records a distinct
# outcome value and releases the arc slot exactly once. Paid amounts stay.
# A settled established end applies the route's lasting, modest policy outcome:
# a country-scope modifier and the same modifier to the captured pilot state
# (whose lower-strata and radicals keys are state-scoped). This is a real
# tradeoff, not a label. No building, resource or foreign consent is created.
# Country-scope outcome: applied to ROOT only, country/interest-group keys only,
# finite 730-day duration.
ywc_lan_gold_apply_route_benefit = {
	if = { limit = { var:ywc_lan_gold_route = 1 } add_modifier = { name = ywc_lan_gold_processing_country days = 730 } }
	else_if = { limit = { var:ywc_lan_gold_route = 2 } add_modifier = { name = ywc_lan_gold_farming_country days = 730 } }
	else_if = { limit = { var:ywc_lan_gold_route = 3 } add_modifier = { name = ywc_lan_gold_monopoly_country days = 730 } }
}

# State-scope outcome: applied only to the captured pilot state, state keys only,
# finite 730-day duration. Never applied to the country and never to any state
# other than the captured original.
ywc_lan_gold_apply_route_benefit_pilot = {
	if = { limit = { ywc_lan_gold_pilot_owned = yes }
		if = { limit = { var:ywc_lan_gold_route = 1 } var:ywc_lan_gold_pilot_state = { add_modifier = { name = ywc_lan_gold_processing_state days = 730 } } }
		else_if = { limit = { var:ywc_lan_gold_route = 2 } var:ywc_lan_gold_pilot_state = { add_modifier = { name = ywc_lan_gold_farming_state days = 730 } } }
		else_if = { limit = { var:ywc_lan_gold_route = 3 } var:ywc_lan_gold_pilot_state = { add_modifier = { name = ywc_lan_gold_monopoly_state days = 730 } } }
	}
}

# Established close: requires the actual 1126 (or 1127 receipt via the same
# stage gate) delivery, the live terminal predicate, and no loss. It never
# fires from a stale cached option because at_1126/at_1127 are re-read here.
ywc_lan_gold_close_established = {
	if = {
		limit = {
			ywc_lan_gold_open = yes NOT = { has_variable = ywc_lan_gold_outcome }
			OR = { ywc_lan_gold_at_1126 = yes ywc_lan_gold_at_1127 = yes }
			ywc_lan_gold_terminal_established = yes
		}
		set_variable = { name = ywc_lan_gold_outcome value = 1 }
		ywc_lan_gold_apply_route_benefit = yes
		ywc_lan_gold_apply_route_benefit_pilot = yes
		ywc_lan_gold_release_once = yes
	}
}

ywc_lan_gold_close_compromise = {

}
