ywc_lan_mine_monthly_review = {
	if = {
		limit = {
			NOT = { has_variable = ywc_lan_mine_established }
			NOT = { has_variable = ywc_lan_mine_compromise }
			NOT = { has_variable = ywc_lan_mine_withdrawn }
		}
		if = {
			limit = { ywc_lan_mine_route_ready = yes }
			if = {
				limit = {
					has_variable = ywc_lan_mine_observation_started
					NOT = { has_variable = ywc_lan_mine_observation_timer }
				}
				set_variable = { name = ywc_lan_mine_six_months_complete value = 1 }
				remove_variable = ywc_lan_mine_observation_started
			}
			if = {
				limit = {
					NOT = { has_variable = ywc_lan_mine_observation_started }
					NOT = { has_variable = ywc_lan_mine_six_months_complete }
				}
				set_variable = { name = ywc_lan_mine_observation_timer value = 1 months = 6 }
				set_variable = { name = ywc_lan_mine_observation_started value = 1 }
			}
		}
		else = {
			remove_variable = ywc_lan_mine_observation_timer
			remove_variable = ywc_lan_mine_observation_started
			remove_variable = ywc_lan_mine_six_months_complete
		}
	}
}
