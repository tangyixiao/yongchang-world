# KOR-local military budget. The opening snapshot uses the official yearly gross
# government income divided by 52; each actual debit is separately guarded.
ywc_kor_capture_cost_base = {
	if = {
		limit = { NOT = { has_variable = ywc_kor_cost_captured } }
		set_variable = { name = ywc_kor_weekly_cost value = yearly_gross_income }
		change_variable = { name = ywc_kor_weekly_cost divide = 52 }
		clamp_variable = { name = ywc_kor_weekly_cost min = 100 max = 1000000000 }
		set_variable = { name = ywc_kor_cost_captured value = 1 }
	}
}

ywc_kor_pay_audit_once = {
	if = {
		limit = {
			net_fixed_income >= 0
			NOT = { has_variable = ywc_kor_audit_paid }
		}
		ywc_kor_capture_cost_base = yes
		if = {
			limit = { gold_reserves >= var:ywc_kor_weekly_cost }
			add_treasury = { value = var:ywc_kor_weekly_cost multiply = -1 }
			set_variable = { name = ywc_kor_audit_paid value = 1 }
		}
	}
}

# Optional local training and veteran transition contracts are finite: each
# selected contract pays at most twelve monthly installments totaling one
# captured week of income. A short lock prevents two JE pulses charging twice.
ywc_kor_pay_selected_contracts = {
	if = {
		limit = {
			OR = { has_variable = ywc_kor_translation_contract_started has_variable = ywc_kor_veteran_contract_started }
			NOT = { has_variable = ywc_kor_contract_month_lock }
		}
		set_variable = { name = ywc_kor_contract_month_lock value = 1 days = 27 }
		ywc_kor_capture_cost_base = yes
		set_variable = { name = ywc_kor_contract_monthly_installment value = var:ywc_kor_weekly_cost }
		change_variable = { name = ywc_kor_contract_monthly_installment multiply = 0.0833333333 }
		if = {
			limit = { has_variable = ywc_kor_translation_contract_started var:ywc_kor_translation_contract_paid_months < 12 net_fixed_income >= 0 gold_reserves >= var:ywc_kor_contract_monthly_installment }
			add_treasury = { value = var:ywc_kor_contract_monthly_installment multiply = -1 }
			change_variable = { name = ywc_kor_translation_contract_paid_months add = 1 }
		}
		if = {
			limit = { has_variable = ywc_kor_veteran_contract_started var:ywc_kor_veteran_contract_paid_months < 12 net_fixed_income >= 0 gold_reserves >= var:ywc_kor_contract_monthly_installment }
			add_treasury = { value = var:ywc_kor_contract_monthly_installment multiply = -1 }
			change_variable = { name = ywc_kor_veteran_contract_paid_months add = 1 }
		}
	}
}

# Keep the old KOR reform thread moving only after this arc has a real result.
# Preserve any chapter-one result already written by the legacy campaign.
ywc_kor_bridge_military_accounts = {
	if = {
		limit = { NOT = { has_variable = ywc_campaign_kor_c1_outcome } }
		if = {
			limit = { var:ywc_v03_kor_v03_kor_military_accounts_outcome = 1 }
			set_variable = { name = ywc_campaign_kor_c1_outcome value = 1 }
			set_variable = { name = ywc_je_flavor_kor_reform_1_resolved value = 1 }
		}
		else_if = {
			limit = { OR = { var:ywc_v03_kor_v03_kor_military_accounts_outcome = 2 var:ywc_v03_kor_v03_kor_military_accounts_outcome = 3 } }
			set_variable = { name = ywc_campaign_kor_c1_outcome value = 2 }
			set_variable = { name = ywc_je_flavor_kor_reform_1_resolved value = 1 }
		}
		else_if = {
			limit = { var:ywc_v03_kor_v03_kor_military_accounts_outcome = 4 }
			set_variable = { name = ywc_campaign_kor_c1_outcome value = 3 }
			set_variable = { name = ywc_je_flavor_kor_reform_1_failed value = 1 }
		}
	}
	if = {
		limit = {
			NOT = { has_journal_entry = ywc_je_flavor_kor_reform_2 }
			NOT = { has_variable = ywc_je_flavor_kor_reform_2_resolved }
			NOT = { has_variable = ywc_je_flavor_kor_reform_2_failed }
		}
		add_journal_entry = { type = ywc_je_flavor_kor_reform_2 }
	}
}
