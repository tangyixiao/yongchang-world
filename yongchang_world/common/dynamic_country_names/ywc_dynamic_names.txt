# Dynamic names are tag-scoped and use only vanilla country-scope triggers.

# KOR and LAN already have vanilla lists in 1.13.11 and keep them; entries for
# one tag merge into the same candidate list (vanilla's own file holds up to
# seven entries per tag) and the highest-priority valid trigger wins.
# MGL needs an explicit override: vanilla 00_dynamic_country_names.txt gives MGL
# "dyn_c_great_mongolian_state" (priority 0, monarchy trigger), which would win
# whenever no mod entry competes. Khalkha entry outranks it at priority 10.
MGL = {
	dynamic_country_name = { name = ywc_dyn_mgl_khalkha adjective = ywc_dyn_mgl_khalkha_adj is_main_tag_only = yes priority = 10 trigger = { exists = scope:actor scope:actor ?= { c:MGL ?= this coa_def_monarchy_flag_trigger = yes } } }
	dynamic_country_name = { name = ywc_dyn_mgl_maritime adjective = ywc_dyn_mgl_maritime_adj is_main_tag_only = yes priority = 20 trigger = { exists = scope:actor scope:actor ?= { c:MGL ?= this ywc_has_maritime_network = yes } } }
}
