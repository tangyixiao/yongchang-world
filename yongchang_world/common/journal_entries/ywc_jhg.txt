# The two routes share maritime capacity but are mutually exclusive once a
# route is started. Autonomy is a real cost, not a free button outcome.

ywc_je_jinghai_maritime_network = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_jinghai_maritime_network_resolved }
                    NOT = { has_variable = ywc_je_jinghai_maritime_network_failed }
                }
                trigger_event = { id = ywc_jhg.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_jinghai_maritime_network_resolved }
    fail = { has_variable = ywc_je_jinghai_maritime_network_failed }
}

                    NOT = { has_variable = ywc_route_jhg_naval_tributary_state_abandonment_cooldown }
                }
                trigger_event = { id = ywc_jhg.4 }
            }
        }
    }
    complete = {
        OR = {
            AND = {
                has_variable = ywc_route_jhg_naval_tributary_state_success

                var:ywc_route_jhg_naval_tributary_state_progress >= 100
            }
        }
    }
    fail = { has_variable = ywc_route_jhg_naval_tributary_state_failure }
}

ywc_je_flavor_jhg_convoy_1 = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            ywc_campaign_tick_national_wait = yes
            if = {
                    limit = {
                            NOT = { has_variable = ywc_campaign_jhg_c1_outcome }
                            OR = {
                                    NOT = { has_variable = ywc_campaign_national_wait }
                                    var:ywc_campaign_national_wait <= 0
                            }
                        }
                        if = {
                                limit = { NOT = { has_variable = ywc_campaign_jhg_c1_prelude } }
                                trigger_event = { id = ywc_jhg.6 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_jhg_c1_e100_done } }
                                trigger_event = { id = ywc_jhg.100 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_jhg_c1_e101_done } }
                                trigger_event = { id = ywc_jhg.101 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_jhg_c1_e102_done } }
                                trigger_event = { id = ywc_jhg.102 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_jhg_c1_e103_done } }
                                trigger_event = { id = ywc_jhg.103 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_jhg_c1_e104_done } }
                                trigger_event = { id = ywc_jhg.104 }
                            }
                    }
                }
            }
    complete = {
        has_variable = ywc_je_flavor_jhg_convoy_1_resolved
    }
    fail = { has_variable = ywc_je_flavor_jhg_convoy_1_failed }
}
