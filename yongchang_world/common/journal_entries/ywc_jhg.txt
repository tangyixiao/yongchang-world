# JHG country content journals.
# The main journal ywc_je_jinghai_autonomy lives in ywc_bootstrap_journal.txt
# (already wired). Auxiliary journals and routes pulse monthly until their
# decision event (ywc_jhg.1-5) is taken. The naval-tributary route raises
# autonomy and, per plan 03, must also raise tension with SHU and fiscal
# cost - consequence effects are tracked for the balance round.

ywc_je_jinghai_maritime_network = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    visible = { is_country = c:JHG }
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_jinghai_maritime_network_resolved }
                    NOT = { has_variable = ywc_je_jinghai_maritime_network_failed }
                }
                trigger_event = { id = ywc_jhg.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_jinghai_maritime_network_resolved }
    fail = { has_variable = ywc_je_jinghai_maritime_network_failed }
}
