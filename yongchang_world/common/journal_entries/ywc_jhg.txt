# JHG country content journals.
# The two routes share maritime capacity but are mutually exclusive once a
# route is started. Autonomy is a real cost, not a free button outcome.

ywc_je_jinghai_maritime_network = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_jinghai_maritime_network_resolved }
                    NOT = { has_variable = ywc_je_jinghai_maritime_network_failed }
                }
                trigger_event = { id = ywc_jhg.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_jinghai_maritime_network_resolved }
    fail = { has_variable = ywc_je_jinghai_maritime_network_failed }
}

ywc_route_jhg_naval_tributary_state = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_route_jhg_naval_tributary_state_success }
                    NOT = { has_variable = ywc_route_jhg_naval_tributary_state_failure }
                    NOT = { has_variable = ywc_route_jhg_naval_tributary_state_abandonment_cooldown }
                }
                trigger_event = { id = ywc_jhg.4 }
            }
        }
    }
    complete = {
        OR = {
            AND = {
                has_variable = ywc_route_jhg_naval_tributary_state_success
                has_variable = ywc_route_jhg_naval_tributary_state_progress
                var:ywc_route_jhg_naval_tributary_state_progress >= 100
            }
        }
    }
    fail = { has_variable = ywc_route_jhg_naval_tributary_state_failure }
}
