# DMG country content journals.
# Journals pulse monthly until their decision event (ywc_dmg.1-5) is taken;
# success/abandon completes them, failure fails them, matching the
# content_catalog route outcomes contract.

ywc_je_dongming_who_inherits_china = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_dongming_who_inherits_china_resolved }
                    NOT = { has_variable = ywc_je_dongming_who_inherits_china_failed }
                }
                trigger_event = { id = ywc_dmg.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_dongming_who_inherits_china_resolved }
    fail = { has_variable = ywc_je_dongming_who_inherits_china_failed }
}

        effect = {
            ywc_campaign_tick_national_wait = yes
            if = {
                    limit = {
                            NOT = { has_variable = ywc_campaign_dmg_c1_outcome }
                            OR = {
                                    NOT = { has_variable = ywc_campaign_national_wait }
                                    var:ywc_campaign_national_wait <= 0
                            }

                        if = {
                                limit = { NOT = { has_variable = ywc_campaign_dmg_c1_prelude } }
                                trigger_event = { id = ywc_dmg.6 }
                            }

                                limit = { NOT = { has_variable = ywc_campaign_dmg_c1_e100_done } }
                                trigger_event = { id = ywc_dmg.100 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_dmg_c1_e101_done } }
                                trigger_event = { id = ywc_dmg.101 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_dmg_c1_e102_done } }
                                trigger_event = { id = ywc_dmg.102 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_dmg_c1_e103_done } }
                                trigger_event = { id = ywc_dmg.103 }
                            }
                        else_if = {
