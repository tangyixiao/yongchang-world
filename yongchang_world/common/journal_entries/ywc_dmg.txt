# DMG country content journals.
# Journals pulse monthly until their decision event (ywc_dmg.1-5) is taken;
# success/abandon completes them, failure fails them, matching the
# content_catalog route outcomes contract.

ywc_je_dongming_who_inherits_china = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    visible = { is_country = c:DMG }
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_dongming_who_inherits_china_resolved }
                    NOT = { has_variable = ywc_je_dongming_who_inherits_china_failed }
                }
                trigger_event = { id = ywc_dmg.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_dongming_who_inherits_china_resolved }
    fail = { has_variable = ywc_je_dongming_who_inherits_china_failed }
}
