# TIB country content journals.
# Journals pulse monthly until their decision event (ywc_tib.1-6) is taken.
# Plan 03: MNG and TIB must not merge into each other at game start.

ywc_je_tibet_highland_without_master = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_tibet_highland_without_master_resolved }
                    NOT = { has_variable = ywc_je_tibet_highland_without_master_failed }
                }
                trigger_event = { id = ywc_tib.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_tibet_highland_without_master_resolved }
    fail = { has_variable = ywc_je_tibet_highland_without_master_failed }
}
