ywc_je_eternal_yongchang = {
    is_shown_in_lobby = { c:SHU ?= THIS }
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_eternal_yongchang_resolved }
                    NOT = { has_variable = ywc_eternal_yongchang_pending }
                }
                set_variable = { name = ywc_eternal_yongchang_pending value = 1 }
                trigger_event = { id = ywc_bootstrap.1 }
            }
        }
    }
    complete = { has_variable = ywc_eternal_yongchang_resolved }
}

ywc_je_jinghai_autonomy = {
    is_shown_in_lobby = { c:JHG ?= THIS }
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_jinghai_autonomy_resolved }
                    NOT = { has_variable = ywc_jinghai_autonomy_failed }
                    NOT = { has_variable = ywc_jinghai_autonomy_pending }
                    NOT = { has_variable = ywc_jinghai_autonomy_defer_cooldown }
                }
                set_variable = { name = ywc_jinghai_autonomy_pending value = 1 }
                trigger_event = { id = ywc_bootstrap.2 }
            }
        }
    }
    complete = {
        AND = {
            has_variable = ywc_jinghai_autonomy_resolved
            has_variable = ywc_jinghai_autonomy_progress
            trigger_if = {
                limit = { has_variable = ywc_jinghai_autonomy_progress }
                var:ywc_jinghai_autonomy_progress >= 100
            }
            trigger_else = { always = no }
        }
    }
    fail = { has_variable = ywc_jinghai_autonomy_failed }
}
