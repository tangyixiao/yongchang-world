# Regional journal shared by the ocean countries (MHG, NMG).
ywc_je_ocean_frontiers = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_ocean_frontiers_resolved }
                    NOT = { has_variable = ywc_ocean_frontiers_failed }
                }
                trigger_event = { id = ywc_shared.2 }
            }
        }
    }
    complete = { has_variable = ywc_ocean_frontiers_resolved }
    fail = { has_variable = ywc_ocean_frontiers_failed }
}
