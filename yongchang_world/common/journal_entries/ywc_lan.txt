# LAN country content journals.
# Journals pulse monthly until their decision event (ywc_lan.1-6) is taken.
# Plan 03: JHG must not colonize Africa; LAN must survive Dutch pressure
# without free bonuses.

ywc_je_lanfang_company_republic = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_lanfang_company_republic_resolved }
                    NOT = { has_variable = ywc_je_lanfang_company_republic_failed }
                }
                trigger_event = { id = ywc_lan.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_lanfang_company_republic_resolved }
    fail = { has_variable = ywc_je_lanfang_company_republic_failed }
}
