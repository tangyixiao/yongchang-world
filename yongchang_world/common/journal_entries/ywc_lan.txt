ywc_je_lanfang_company_republic = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_lanfang_company_republic_resolved }
                    NOT = { has_variable = ywc_je_lanfang_company_republic_failed }
                }
                trigger_event = { id = ywc_lan.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_lanfang_company_republic_resolved }
    fail = { has_variable = ywc_je_lanfang_company_republic_failed }
}

ywc_je_flavor_lan_charter_1 = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            ywc_campaign_tick_national_wait = yes
            if = {
                    limit = {
                            NOT = { has_variable = ywc_campaign_lan_c1_outcome }
                            OR = {
                                    NOT = { has_variable = ywc_campaign_national_wait }
                                    var:ywc_campaign_national_wait <= 0
                            }
                        }
                        if = {
                                limit = { NOT = { has_variable = ywc_campaign_lan_c1_prelude } }
                                trigger_event = { id = ywc_lan.7 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_lan_c1_e100_done } }
                                trigger_event = { id = ywc_lan.100 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_lan_c1_e101_done } }
                                trigger_event = { id = ywc_lan.101 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_lan_c1_e102_done } }
                                trigger_event = { id = ywc_lan.102 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_lan_c1_e103_done } }
                                trigger_event = { id = ywc_lan.103 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_lan_c1_e104_done } }
                                trigger_event = { id = ywc_lan.104 }
                            }
                    }
                }
            }
    complete = {
        has_variable = ywc_je_flavor_lan_charter_1_resolved

}
