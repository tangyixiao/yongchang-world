ywc_je_dlc_compatibility = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	group = je_group_internal_affairs
	on_monthly_pulse = {
		effect = {
			if = {
				limit = { NOT = { has_variable = ywc_dlc_base_path } }
				ywc_apply_dlc_compatibility = yes
			}
		}
	}
	complete = {
		has_variable = ywc_dlc_base_path
		OR = {
			NOT = { has_dlc_feature = ep1_content }
			has_variable = ywc_dlc_ep1_active
		}
		OR = {
			NOT = { has_dlc_feature = mp1_content }
			has_variable = ywc_dlc_mp1_active
		}
		OR = {
			NOT = { has_dlc_feature = ep2_content }
			has_variable = ywc_dlc_ep2_active
		}
	}
}
