# Crisis journals for the 2026-09-25 cross-country crises. Each journal
# is added to its holder when the gating chapter settles; the pulse
# advances one stage at a time and every stage answers on its own country.

ywc_je_crisis_suzerainty = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_foreign_affairs
    on_monthly_pulse = {
        effect = {
            ywc_campaign_tick_dialogue_cooldown = yes
            if = {
                    limit = {
                            OR = {
                                    NOT = { has_variable = ywc_campaign_dialogue_cooldown }
                                    var:ywc_campaign_dialogue_cooldown <= 0
                            }
                        }
                        ywc_crisis_pulse_suzerainty = yes
                    }
                }
            }
    complete = { has_variable = ywc_campaign_crisis1_outcome }
}
