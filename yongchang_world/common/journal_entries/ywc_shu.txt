# SHU country content journals.
# The main journal ywc_je_yongchang_century is deliberately NOT defined here:
# SHU's playable main journal is ywc_je_eternal_yongchang in
# ywc_bootstrap_journal.txt (a catalog naming drift, recorded in docs).
# Auxiliary journals and routes pulse monthly until their decision event
# (ywc_shu.1-5) is taken.

ywc_je_yongchang_century = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    visible = { is_country = c:SHU }
    complete = { has_variable = ywc_je_yongchang_century_resolved }
}

ywc_je_shu_scholar_merchants = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    visible = { is_country = c:SHU }
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_shu_scholar_merchants_resolved }
                    NOT = { has_variable = ywc_je_shu_scholar_merchants_failed }
                }
                trigger_event = { id = ywc_shu.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_shu_scholar_merchants_resolved }
    fail = { has_variable = ywc_je_shu_scholar_merchants_failed }
}
