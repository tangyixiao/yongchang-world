# SHU country content journals.
# The main line and the two routes use explicit progress variables. A route
# cannot complete from a single monthly decision: the player must build five
# milestones, then qualify for the conclusion option.

ywc_je_yongchang_century = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_yongchang_century_resolved }
                    NOT = { has_variable = ywc_je_yongchang_century_failed }
                }
                trigger_event = { id = ywc_shu.6 }
            }
        }
    }
    complete = {
        AND = {
            has_variable = ywc_je_yongchang_century_resolved
            has_variable = ywc_je_yongchang_century_progress
            var:ywc_je_yongchang_century_progress >= 100
        }
    }
    fail = { has_variable = ywc_je_yongchang_century_failed }
}

ywc_je_flavor_shu_court_1 = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            ywc_campaign_tick_national_wait = yes
            if = {
                    limit = {
                            NOT = { has_variable = ywc_campaign_shu_c1_outcome }
                            OR = {
                                    NOT = { has_variable = ywc_campaign_national_wait }
                                    var:ywc_campaign_national_wait <= 0
                            }
                        }
                        if = {
                                limit = { NOT = { has_variable = ywc_campaign_shu_c1_prelude } }
                                trigger_event = { id = ywc_shu.7 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_shu_c1_e100_done } }
                                trigger_event = { id = ywc_shu.100 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_shu_c1_e101_done } }
                                trigger_event = { id = ywc_shu.101 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_shu_c1_e102_done } }
                                trigger_event = { id = ywc_shu.102 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_shu_c1_e103_done } }
                                trigger_event = { id = ywc_shu.103 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_shu_c1_e104_done } }
                                trigger_event = { id = ywc_shu.104 }
                            }
                    }
                }
            }
    complete = {
        has_variable = ywc_je_flavor_shu_court_1_resolved

}
