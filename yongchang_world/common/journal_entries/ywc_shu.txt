# SHU country content journals.
# SHU holds two main journals: ywc_je_yongchang_century (added in
# content_starts, wired to ywc_shu.6 below) and ywc_je_eternal_yongchang
# (added from ywc_shu.txt history, wired in ywc_bootstrap_journal.txt).
# Auxiliary journals and routes pulse monthly until their decision event
# (ywc_shu.1-6) is taken.

ywc_je_yongchang_century = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    visible = { is_country = c:SHU }
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_yongchang_century_resolved }
                    NOT = { has_variable = ywc_je_yongchang_century_failed }
                }
                trigger_event = { id = ywc_shu.6 }
            }
        }
    }
    complete = { has_variable = ywc_je_yongchang_century_resolved }
    fail = { has_variable = ywc_je_yongchang_century_failed }
}
