# MNG country content journals.
# Journals pulse monthly until their decision event (ywc_mng.1-6) is taken.
# The main journal is a direction decision: either answer completes it and
# records whether the khans leaned south (Shun) or north (Russia).

ywc_je_mongol_south_north = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_mongol_south_north_resolved }
                }
                trigger_event = { id = ywc_mng.1 }
            }
        }
    }
    complete = {
        has_variable = ywc_je_mongol_south_north_resolved
        OR = {
            has_variable = ywc_mng_south_alignment
            has_variable = ywc_mng_north_alignment
        }
    }
}

ywc_route_mng_southern_trade = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"

    on_monthly_pulse = {

                }
                trigger_event = { id = ywc_mng.5 }
            }
        }
    }
    complete = {
        OR = {
            AND = {
                has_variable = ywc_route_mng_southern_trade_success
                has_variable = ywc_route_mng_southern_trade_progress
                var:ywc_route_mng_southern_trade_progress >= 100
            }
        }
    }
    fail = { has_variable = ywc_route_mng_southern_trade_failure }
}
