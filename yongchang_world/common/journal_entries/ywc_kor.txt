# KOR country content journals.
# Journals pulse monthly until their decision event (ywc_kor.1-6) is taken.

ywc_je_korea_who_inherits_china = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_korea_who_inherits_china_resolved }
                    NOT = { has_variable = ywc_je_korea_who_inherits_china_failed }
                }
                trigger_event = { id = ywc_kor.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_korea_who_inherits_china_resolved }
    fail = { has_variable = ywc_je_korea_who_inherits_china_failed }
}

ywc_route_kor_small_china = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_route_kor_small_china_success }
                    NOT = { has_variable = ywc_route_kor_small_china_failure }
                    OR = {
                        NOT = { has_variable = ywc_route_kor_small_china_abandon }
                        NOT = { has_variable = ywc_route_kor_small_china_abandonment_cooldown }
                    }
                }
                trigger_event = { id = ywc_kor.5 }
            }
        }
    }
    complete = {
        OR = {
            AND = {
                has_variable = ywc_route_kor_small_china_success
                has_variable = ywc_route_kor_small_china_progress
                var:ywc_route_kor_small_china_progress >= 100
            }
        }
    }
    fail = { has_variable = ywc_route_kor_small_china_failure }
}
