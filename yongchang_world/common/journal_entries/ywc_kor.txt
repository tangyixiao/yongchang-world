# KOR country content journals.
# Journals pulse monthly until their decision event (ywc_kor.1-6) is taken.

ywc_je_korea_who_inherits_china = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_korea_who_inherits_china_resolved }
                    NOT = { has_variable = ywc_je_korea_who_inherits_china_failed }
                }
                trigger_event = { id = ywc_kor.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_korea_who_inherits_china_resolved }
    fail = { has_variable = ywc_je_korea_who_inherits_china_failed }
}

ywc_je_flavor_kor_reform_1 = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            ywc_campaign_tick_national_wait = yes
            if = {
                    limit = {
                            NOT = { has_variable = ywc_campaign_kor_c1_outcome }
                            OR = {
                                    NOT = { has_variable = ywc_campaign_national_wait }
                                    var:ywc_campaign_national_wait <= 0
                            }
                        }
                        if = {
                                limit = { NOT = { has_variable = ywc_campaign_kor_c1_prelude } }
                                trigger_event = { id = ywc_kor.7 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_kor_c1_e100_done } }
                                trigger_event = { id = ywc_kor.100 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_kor_c1_e101_done } }
                                trigger_event = { id = ywc_kor.101 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_kor_c1_e102_done } }
                                trigger_event = { id = ywc_kor.102 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_kor_c1_e103_done } }
                                trigger_event = { id = ywc_kor.103 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_kor_c1_e104_done } }
                                trigger_event = { id = ywc_kor.104 }
                            }
                    }
                }
            }
    complete = {
        has_variable = ywc_je_flavor_kor_reform_1_resolved
    }
    fail = { has_variable = ywc_je_flavor_kor_reform_1_failed }
}
