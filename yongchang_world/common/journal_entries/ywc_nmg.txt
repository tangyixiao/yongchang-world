# ywc_nmg_autonomy_negotiation diplomatic action (see

# journals and routes pulse monthly until their decision event
# (ywc_nmg.1-5) is taken.

    icon = "gfx/interface/icons/event_icons/event_portrait.dds"

    complete = { has_variable = ywc_je_new_ming_mexican_chain_resolved }
}

ywc_je_nmg_mexican_federation = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_nmg_mexican_federation_resolved }
                    NOT = { has_variable = ywc_je_nmg_mexican_federation_failed }
                }
                trigger_event = { id = ywc_nmg.3 }
            }
        }
    }
    complete = { has_variable = ywc_je_nmg_mexican_federation_resolved }
    fail = { has_variable = ywc_je_nmg_mexican_federation_failed }
}

    fail = { has_variable = ywc_route_nmg_catholic_autonomy_failure }
}
