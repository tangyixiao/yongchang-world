# OIR country content journals.
# Journals pulse monthly until their decision event (ywc_oir.1-6) is taken.
# Plan 03: the succession crisis and Russian pressure must stay real, and
# survival through reform must remain possible without any expansion.

ywc_je_oir_dzungar_legacy = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_oir_dzungar_legacy_resolved }
                    NOT = { has_variable = ywc_je_oir_dzungar_legacy_failed }
                }
                trigger_event = { id = ywc_oir.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_oir_dzungar_legacy_resolved }
    fail = { has_variable = ywc_je_oir_dzungar_legacy_failed }
}
