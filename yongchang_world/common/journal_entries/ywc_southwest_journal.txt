# Southwest regional journals (v0.2 六六大顺).
# The old regional journal ywc_je_highland_without_master was removed:
# it duplicated TIB's main journal ywc_je_tibet_highland_without_master
# and was visible to every country. TIB is wired through its own file.

ywc_je_sw_ljg = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = { NOT = { has_variable = ywc_sw_ljg_e1_done } }
                trigger_event = { id = ywc_ljg.1 }
            }
            else_if = {
                limit = { NOT = { has_variable = ywc_sw_ljg_e2_done } }
                trigger_event = { id = ywc_ljg.2 }
            }
            else_if = {
                limit = { NOT = { has_variable = ywc_sw_ljg_e3_done } }
                trigger_event = { id = ywc_ljg.3 }
            }
        }
    }
    complete = { has_variable = ywc_sw_ljg_resolved }
    fail = { has_variable = ywc_sw_ljg_failed }
}
