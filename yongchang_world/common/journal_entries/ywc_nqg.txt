# NQG country content journals.
# Journals pulse monthly until their decision event (ywc_nqg.1-5) is taken;
# routes follow the content_catalog three-outcome contract. The Russian
# choice is a decision journal: either answer completes it and records the
# path taken. Expansion must not turn NQG into a Manchu great power
# (catalog: forbidden become_great_power / mass_settler_colonization).

ywc_je_northern_qing_exile = {
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    group = je_group_internal_affairs
    on_monthly_pulse = {
        effect = {
            if = {
                limit = {
                    NOT = { has_variable = ywc_je_northern_qing_exile_resolved }
                    NOT = { has_variable = ywc_je_northern_qing_exile_failed }
                }
                trigger_event = { id = ywc_nqg.1 }
            }
        }
    }
    complete = { has_variable = ywc_je_northern_qing_exile_resolved }
    fail = { has_variable = ywc_je_northern_qing_exile_failed }
}

        effect = {
            ywc_campaign_tick_national_wait = yes
            if = {
                    limit = {
                            NOT = { has_variable = ywc_campaign_nqg_c1_outcome }
                            OR = {
                                    NOT = { has_variable = ywc_campaign_national_wait }
                                    var:ywc_campaign_national_wait <= 0
                            }

                        if = {
                                limit = { NOT = { has_variable = ywc_campaign_nqg_c1_prelude } }
                                trigger_event = { id = ywc_nqg.6 }
                            }
                        else_if = {
                                limit = { NOT = { has_variable = ywc_campaign_nqg_c1_e100_done } }

                                trigger_event = { id = ywc_nqg.101 }
