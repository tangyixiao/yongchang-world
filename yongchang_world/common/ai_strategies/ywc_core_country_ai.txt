# Active country-specific diplomatic strategies for the ten Yongchang states.
#
# These are assigned by ywc_on_game_started_after_lobby. The route variables
# are deliberately read here instead of granting free troops or money: a
# country becomes bolder only after it has invested in a route, while an active
# crisis, subject status, or bankruptcy pushes the AI back toward survival.

# SHU: domestic finance, northeast security, and limited subject relations.
ai_strategy_ywc_shu = {
	icon = "gfx/interface/icons/ai_strategy_icons/maintain_power_balance.dds"
	type = diplomatic
	desired_tax_level = low
	min_tax_level = low
	max_tax_level = high
	undesirable_infamy_level = { value = 35 }
	unacceptable_infamy_level = { value = 70 }
	diplomatic_play_neutrality = {
		value = 5
		if = { limit = { is_active_in_diplomatic_play = yes } add = 20 }
		if = { limit = { has_modifier = declared_bankruptcy } add = 20 }
		if = { limit = { is_subject = yes } add = 10 }
	}
	diplomatic_play_boldness = {
		value = -15
		if = { limit = { has_variable = ywc_route_shu_bureaucratic_integration_progress } add = 10 }
		if = { limit = { has_variable = ywc_route_shu_customs_reform_progress } add = 10 }
		if = { limit = { is_active_in_diplomatic_play = yes } add = -15 }
	}
	recklessness = {
		value = -0.20
		if = { limit = { has_modifier = declared_bankruptcy } add = -0.20 }
		if = { limit = { has_variable = ywc_route_shu_bureaucratic_integration_progress var:ywc_route_shu_bureaucratic_integration_progress >= 80 } add = 0.10 }
	}
	aggression = { value = -0.10 if = { limit = { is_subject = yes } add = -0.15 } }
	colonization_rights = { value = 0 }
	building_group_weights = { bg_construction = 1.20 bg_manufacturing = 1.15 bg_mining = 0.90 }
	goods_stances = { tools = { stance = wants_high_supply } iron = { stance = wants_high_supply } coal = { stance = wants_high_supply } }
	possible = { always = yes }
	weight = {
		value = 20
		if = { limit = { has_modifier = declared_bankruptcy } multiply = 0.5 }
		if = { limit = { has_variable = ywc_route_shu_bureaucratic_integration_progress } add = 10 }
	}
}
