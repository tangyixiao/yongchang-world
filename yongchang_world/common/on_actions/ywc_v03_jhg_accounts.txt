on_monthly_pulse_country = {
	effect = {
		if = {
			limit = {
				c:JHG ?= this
				OR = {
					var:ywc_jhg_escort_state ?= 1
					var:ywc_jhg_escort_state ?= 2
					has_variable = ywc_jhg_escort_contract_generation
				}
			}
			ywc_jhg_escort_monthly = yes
		}
		if = {
			limit = {
				c:JHG ?= this
				OR = { var:ywc_jhg_share_state ?= 1 var:ywc_jhg_share_state ?= 2 has_variable = ywc_jhg_share_contract_generation }
			}
			ywc_jhg_share_monthly = yes
		}
	}
}

# Installed 00_code_on_actions.txt declares both callbacks with Root = country.
# Only unsigned proposals are invalidated; signed costs remain in their ledger.
on_become_subject = {
	effect = {
		if = {
			limit = { c:JHG ?= this }
			ywc_jhg_escort_cancel_unsigned = yes
			ywc_jhg_share_cancel_unsigned = yes
		}
	}
}

on_become_independent = {
	effect = {
		if = {
			limit = { c:JHG ?= this }
			ywc_jhg_escort_cancel_unsigned = yes
			ywc_jhg_share_cancel_unsigned = yes
		}
	}
}
