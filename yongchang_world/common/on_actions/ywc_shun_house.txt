# Character-scoped native hooks preserve the scenario's Li house.
on_new_ruler = { on_actions = { ywc_shu_keep_li_surname } }
on_heir_born = { on_actions = { ywc_shu_keep_li_surname } }
ywc_shu_keep_li_surname = {
    effect = {
        if = {
            limit = { owner = { c:SHU ?= this } }
            set_last_name = Li
        }
    }
}
