# Regional MHG (F63) startup hook. Additive to the shared lobby hook; guarded
# so a world without MHG never dereferences a missing tag.
on_game_started_after_lobby = { on_actions = { ywc_v03_regional_mekong_after_lobby } }

ywc_v03_regional_mekong_after_lobby = {
	effect = {
		if = {
			limit = { exists = c:MHG }
			c:MHG = {
				if = {
					limit = {
						is_country_alive = yes
						NOT = { has_variable = ywc_mhg_land_waterway_started }
						NOT = { has_journal_entry = ywc_je_v03_mhg_land_waterway }
					}
					add_journal_entry = { type = ywc_je_v03_mhg_land_waterway }
				}
			}
		}
	}
}
