on_monthly_pulse_country = { on_actions = { ywc_v03_kor_accounts_monthly } }

ywc_v03_kor_accounts_monthly = {
 effect = {
  if = {
   limit = {
    c:KOR ?= this
    has_variable = ywc_v03_kor_v03_kor_military_accounts_settled
    OR = {
     AND = { has_variable = ywc_kor_translation_contract_started var:ywc_kor_translation_contract_paid_months < 12 }
     AND = { has_variable = ywc_kor_veteran_contract_started var:ywc_kor_veteran_contract_paid_months < 12 }
    }
   }
   ywc_kor_pay_selected_contracts = yes
  }
  if = {
   limit = { c:KOR ?= this OR = { var:ywc_kor_instructor_state ?= 1 has_variable = ywc_kor_instructor_contract_generation } }
   ywc_kor_instructor_monthly = yes
  }
  if = {
   limit = {
    c:KOR ?= this
    OR = {
     var:ywc_kor_procurement_state ?= 1
     var:ywc_kor_procurement_state ?= 2
     has_variable = ywc_kor_procurement_contract_generation
    }
   }
   ywc_kor_procurement_monthly = yes
  }
 }
}
