# The normal new-campaign hook; deliberately no load hook is assumed.
on_game_started_after_lobby = { on_actions = { ywc_v03_initialize_after_lobby } }

ywc_v03_initialize_after_lobby = {
	effect = {
		every_country = {
			limit = {
				# Exists-check comparisons: a tag absent from the current world must
				# not raise "Event target link c returned an unset scope".
				OR = {
					c:SHU ?= this
					c:DMG ?= this
					c:KOR ?= this
					c:JHG ?= this
					c:LAN ?= this
					c:NMG ?= this
					c:NQG ?= this
					c:OIR ?= this
					c:MGL ?= this
					c:TIB ?= this
				}
			}
			ywc_v03_initialize_country = yes
		}
	}
}
