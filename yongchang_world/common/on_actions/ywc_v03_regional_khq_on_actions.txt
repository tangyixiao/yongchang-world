# Regional KHQ (F02) startup hook. Additive to the shared lobby hook; guarded
# so a world without KHQ never dereferences a missing tag. Only journal A opens
# at startup; the league compact journal opens from a settled banner-land line.
on_game_started_after_lobby = { on_actions = { ywc_v03_regional_khq_after_lobby } }

ywc_v03_regional_khq_after_lobby = {
	effect = {
		if = {
			limit = { exists = c:KHQ }
			c:KHQ = {
				if = {
					limit = {
						is_country_alive = yes
						NOT = { has_variable = ywc_khq_a_started }
						NOT = { has_journal_entry = ywc_je_v03_khq_a }
					}
					add_journal_entry = { type = ywc_je_v03_khq_a }
				}
			}
		}
	}
}
